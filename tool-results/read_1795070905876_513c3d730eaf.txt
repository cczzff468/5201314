   186→/** LLM：OpenAI 兼容 chat/completions（非流式），跟随手机 App「API 设置」 */
   187→function llmCandidates(baseUrl) {
   188→  var b = String(baseUrl || '').replace(/\/+$/, '');
   189→  if (!b) return [];
   190→  if (/\/chat\/completions$/.test(b)) return [b];
   191→  if (/\/v\d+$/.test(b)) return [b + '/chat/completions'];
   192→  return [b + '/v1/chat/completions', b + '/chat/completions'];
   193→}

   195→async function callLlm(cfg, messages) {
   196→  var llm = cfg.llm || {};
   197→  var cands = llmCandidates(llm.baseUrl);
   198→  var lastErr = '未配置模型 API（在手机 App 设置 → API 设置里填写后重新同步）';
   199→  for (var i = 0; i < cands.length; i++) {
   200→    try {
   201→      var r = await fetch(cands[i], {
   202→        method: 'POST',
   203→        headers: {
   204→          'Content-Type': 'application/json',
   205→          Authorization: llm.apiKey ? 'Bearer ' + llm.apiKey : '',
   206→        },
   207→        body: JSON.stringify({ model: llm.model || 'gpt-4o-mini', messages: messages, max_tokens: 500 }),
   208→        // v8：原来没有超时——模型 API 挂起时整条收发循环被拖死（消息永远不回）；75s 与
   209→        // /api/chat 的非流式窗口对齐，超时进 lastErr，外层 handleWxMsg 的重试链路接着兑
   210→        signal: AbortSignal.timeout(75000),
   211→      });
   212→      if (!r.ok) { lastErr = 'LLM ' + r.status; continue; }
   213→      var j = await r.json();
   214→      var text = j && j.choices && j.choices[0] && j.choices[0].message && j.choices[0].message.content;
   215→      if (typeof text === 'string' && text.trim()) return text.trim();
   216→      lastErr = 'LLM 空回复';
   217→    } catch (e) { lastErr = String(e); }
   218→  }
   219→  throw new Error(lastErr);
   220→}
