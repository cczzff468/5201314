    56→/** 接入时间格式化（YYYY-MM-DD HH:mm） */
    57→function fmtConnected(t: number): string {
    58→  const d = new Date(t);
    59→  const p = (n: number) => String(n).padStart(2, '0');
    60→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`;
    61→}
