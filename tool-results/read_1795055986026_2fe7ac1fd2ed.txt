    61→/** 接入时间格式化（YYYY-MM-DD HH:mm） */
    62→function fmtConnected(t: number): string {
    63→  const d = new Date(t);
    64→  const p = (n: number) => String(n).padStart(2, '0');
    65→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`;
    66→}
