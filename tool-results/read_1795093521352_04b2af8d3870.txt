    27→// ---------------- v13 本地 TTS（QQ 语音条零配置合成：z-ai SDK，wav） ----------------
    28→
    29→/** z-ai 可用音色（不认识的音色名一律回 tongtong——运行包里存的是 OpenAI 音色名时也不报错） */
    30→const ZAI_VOICES = ['tongtong', 'chuichui', 'xiaochen', 'jam', 'kazi', 'douji', 'luodo'];
    31→let zaiInstance: any | null = null;
    32→
    33→/** 本地合成语音（POST /tts，OpenAI 兼容形状 {input, voice, speed}）→ wav 二进制。
    34→ *  z-ai-web-dev-sdk 仅后端使用（本 mini-service 是后端）；SDK 缺失/合成失败返回 5xx，
    35→ *  云端核心拿到非 200 自动降级纯文字（不炸收发主流程）——QQ 语音条零配置可用。 */
    36→async function handleLocalTts(req: Request): Promise<Response> {
    37→  let body: Record<string, any> = {};
    38→  try {
    39→    body = (await req.json()) as Record<string, any>;
    40→  } catch {
    41→    return Response.json({ error: 'bad json' }, { status: 400 });
    42→  }
    43→  const text = String(body.input || body.text || '').trim().slice(0, 1000);
    44→  if (!text) return Response.json({ error: 'empty text' }, { status: 400 });
    45→  const voiceRaw = String(body.voice || '').trim();
    46→  const voice = ZAI_VOICES.includes(voiceRaw) ? voiceRaw : 'tongtong';
    47→  const speedRaw = Number(body.speed);
    48→  const speed = Number.isFinite(speedRaw) ? Math.min(2, Math.max(0.5, speedRaw)) : 1;
    49→  try {
    50→    if (!zaiInstance) {
    51→      const mod: any = await import('z-ai-web-dev-sdk');
    52→      const ZAI = mod.default ?? mod;
    53→      zaiInstance = await ZAI.create();
    54→    }
    55→    const res = await zaiInstance.audio.tts.create({
    56→      input: text,
    57→      voice,
    58→      speed,
    59→      response_format: 'wav',
    60→      stream: false,
    61→    });
    62→    const buf = new Uint8Array(await res.arrayBuffer());
    63→    if (!buf.length) return Response.json({ error: 'empty audio' }, { status: 502 });
    64→    return new Response(buf, {
    65→      headers: { 'Content-Type': 'audio/wav', 'Content-Length': String(buf.length) },
    66→    });
    67→  } catch (e) {
    68→    zaiInstance = null; // 失败重置单例（下汰重试）
    69→    return Response.json({ error: 'tts failed: ' + String((e as Error)?.message || e).slice(0, 80) }, { status: 503 });
    70→  }
    71→}

    73→// ---------------- v15 本地生图（AI 发照片：z-ai SDK 生图 → sharp 压成 JPEG） ----------------
    74→
    75→/** z-ai 可用生图尺寸（发照片场景 1:1 最自然；云端核心传 prompt，这里选尺寸） */
    76→const ZAI_IMG_SIZE = '1024x1024';
    77→let sharpMod: any | null = null;
    78→
    79→/** 本地生成一张照片（POST /imggen，形状 {prompt}）→ {ok, b64, mime}。
    80→ *  z-ai 生 PNG（1024²照片常 1~2MB）→ sharp 转 JPEG q82（百 KB 级）——两通道富媒体
    81→ *  上传都限 2MB（QQ files / 微信 CDN），不压缩必超。sharp 缺失时 PNG 本体 ≤1.6MB 才回，
    82→ *  否则 5xx（云端核心拿到非 200 自动降级：标记剥除不发图，不炸收发主流程）。
    83→ *  z-ai-web-dev-sdk 仅后端使用（本 mini-service 是后端）。 */
    84→async function handleLocalImggen(req: Request): Promise<Response> {
    85→  let body: Record<string, any> = {};
    86→  try {
    87→    body = (await req.json()) as Record<string, any>;
    88→  } catch {
    89→    return Response.json({ error: 'bad json' }, { status: 400 });
    90→  }
    91→  const prompt = String(body.prompt || '').trim().slice(0, 200);
    92→  if (!prompt) return Response.json({ error: 'empty prompt' }, { status: 400 });
    93→  try {
    94→    if (!zaiInstance) {
    95→      const mod: any = await import('z-ai-web-dev-sdk');
    96→      const ZAI = mod.default ?? mod;
    97→      zaiInstance = await ZAI.create();
    98→    }
    99→    const res = await zaiInstance.images.generations.create({
   100→      prompt,
   101→      size: ZAI_IMG_SIZE,
   102→    });
   103→    const b64png: string = String(res?.data?.[0]?.base64 || '');
   104→    if (!b64png) return Response.json({ error: 'empty image' }, { status: 502 });
   105→    // 压成 JPEG：QQ/微信富媒体上传 ≤2MB，PNG 生图常超；sharp 不可用时 PNG 本体 ≤1.6MB 才回
   106→    try {
   107→      if (!sharpMod) sharpMod = await import('sharp');
   108→      const buf = await sharpMod(Buffer.from(b64png, 'base64')).jpeg({ quality: 82 }).toBuffer();
   109→      if (buf.length && buf.length < Buffer.from(b64png, 'base64').length) {
   110→        return Response.json({ ok: true, b64: buf.toString('base64'), mime: 'image/jpeg' });
   111→      }
   112→    } catch {
   113→      /* sharp 不可用：走 PNG 本体兑底 */
   114→    }
   115→    if (Buffer.from(b64png, 'base64').length <= 1600 * 1024) {
   116→      return Response.json({ ok: true, b64: b64png, mime: 'image/png' });
   117→    }
   118→    return Response.json({ error: 'image too large' }, { status: 503 });
   119→  } catch (e) {
   120→    zaiInstance = null; // 失败重置单例（下次重试）
   121→    return Response.json({ error: 'imggen failed: ' + String((e as Error)?.message || e).slice(0, 80) }, { status: 503 });
   122→  }
   123→}

   270→function scheduleReconnect(gen: number): void {
   271→  if (!rt || rt.gen !== gen || wantClose) return;
   272→  rt.qq.state = 'connecting';
   273→  rt.qq.reconnects += 1;
   274→  reconnectDelay = Math.min(Math.round(reconnectDelay * 1.6), 30000);
   275→  if (reconnectTimer) clearTimeout(reconnectTimer);
   276→  reconnectTimer = setTimeout(() => {
   277→    void connectQq(gen);
   278→  }, reconnectDelay);
   279→}

   281→async function connectQq(gen: number): Promise<void> {
   282→  if (!rt || rt.gen !== gen || wantClose) return;
   283→  // v11 并发防抖：一轮连接在飞（config→token→网关地址→WS 握手）就不再开第二条
   284→  if (qqConnecting) return;
   285→  // v11 僵尸连接防抖：上一条 socket 还活着/还在握手就不再开新的——旧版会直接把 ws 变量
   286→  // 指到新 socket，旧 socket 依旧收事件但心跳发到死 socket 上，状态永远「connecting」
   287→  if (ws && (ws.readyState === 0 || ws.readyState === 1)) return;
   288→  qqConnecting = true;
   289→  try {
   290→  const q = rt.qq;
   291→  try {
   292→    // 凭证来自用户 Supabase 运行包（loadConfig 60s 缓存；换绑/改密钥 60s 内跟上）
   293→    const cfg = await rt.core.loadConfig(true);
   294→    if (!cfg.qq || !cfg.qq.appId) {
   295→      q.state = 'disabled';
   296→      q.lastError = '';
   297→      // v10：没凭证不是终态——手机 App 接入 QQ / 自动同步运行包后这里要自己亮起来。
   298→      // 30 秒后再查一次（fetch 一次 config.json，重活轻）；旧版直接 return，接入后
   299→      // 网关永远不知道，QQ 通道永远「正在连接」，用户实测踩坑。
   300→      if (reconnectTimer) clearTimeout(reconnectTimer);
   301→      reconnectTimer = setTimeout(() => {
   302→        void connectQq(gen);
   303→      }, 30000);
   304→      return;
   305→    }
   306→    q.state = 'connecting';
   307→    const token = await rt.core.qqToken(cfg);
   308→    // 取 WSS 网关地址（官方：GET /gateway，鉴权 QQBot <access_token>）；
   309→    // v10 加超时：原来无超时，QQ 网关挂起时 connectQq 卡死 → QQ 通道永远 connecting
   310→    const gwr = await fetch(QQ_GATEWAY_API, {
   311→      headers: { Authorization: 'QQBot ' + token },
   312→      signal: AbortSignal.timeout(20000),
   313→    });
   314→    if (!gwr.ok) {
   315→      q.state = 'error';
   316→      q.lastError = `QQ 网关地址获取失败 ${gwr.status}`;
   317→      scheduleReconnect(gen);
   318→      return;
   319→    }
   320→    const gwj = (await gwr.json().catch(() => null)) as { url?: string } | null;
   321→    const wssUrl = gwj?.url;
   322→    if (!wssUrl) {
   323→      q.state = 'error';
   324→      q.lastError = 'QQ 网关地址获取失败：回包无 url';
   325→      scheduleReconnect(gen);
   326→      return;
   327→    }
   328→    wantClose = false;
   329→    ws = new WebSocket(wssUrl);
   330→    // v14 握手看门狗：WS 握手挂在网络里（onopen/onclose 都不来）时旧版会永远停在
   331→    // 「connecting」——状态卡死、不重试（onclose 也不会来）——「QQ 接入过一会断开」
   332→    // 的根因之一。20s 内还没进入 ready 就强拆走重连（按 socket 打标，不误伤新一轮尝试）。
   333→    const sock = ws;
   334→    setTimeout(() => {
   335→      if (!rt || rt.gen !== gen || wantClose) return;
   336→      if (ws !== sock) return; // 已被新一轮尝试替换：不管旧连接
   337→      if (rt.qq.state === 'connecting') {
   338→        console.log('[im-gateway] QQ WS 握手看门狗触发（20s 未就绪）→ 强拆重连');
   339→        try {
   340→          ws?.close();
   341→        } catch {
   342→          /* 忽略 */
   343→        }
   344→        scheduleReconnect(gen);
   345→      }
   346→    }, 20000);
   347→    ws.onmessage = (ev: MessageEvent) => {
   348→      if (!rt || rt.gen !== gen || wantClose) return;
   349→      try {
   350→        void handleWsFrame(JSON.parse(String(ev.data)) as Record<string, any>, gen, token);
   351→      } catch {
   352→        /* 非 JSON 帧忽略 */
   353→      }
   354→    };
   355→    ws.onclose = (ev: CloseEvent) => {
   356→      stopHbTimer();
   357→      console.log('[im-gateway] QQ WS close:', ev.code, ev.reason?.slice(0, 80), 'gen', gen);
   358→      if (rt && rt.gen === gen && !wantClose) scheduleReconnect(gen);
   359→    };
   360→    ws.onerror = () => {
   361→      /* onclose 兜底重连；这里只记状态 */
   362→      if (rt && rt.gen === gen) rt.qq.state = rt.qq.sessionId ? 'connecting' : 'error';
   363→    };
   364→  } catch (e) {
   365→    q.state = 'error';
   366→    q.lastError = String((e as Error)?.message || e).slice(0, 160);
   367→    scheduleReconnect(gen);
   368→  }
   369→  } finally {
   370→    qqConnecting = false;
   371→  }
   372→}

   374→async function handleWsFrame(f: Record<string, any>, gen: number, token: string): Promise<void> {
   375→  if (!rt || rt.gen !== gen) return;
   376→  const q = rt.qq;
   377→  const op = Number(f.op);
   378→  if (typeof f.s === 'number' && f.s > 0) q.lastSeq = f.s;
   379→
   380→  if (op === 10) {
   381→    // Hello：心跳周期 + 登录鉴权（有会话则 Resume 断点续传，官方补发遗漏事件）
   382→    hbInterval = Number(f.d?.heartbeat_interval) || 41250;
   383→    stopHbTimer();
   384→    q.hbOk = true;
   385→    hbTimer = setInterval(() => {
   386→      if (!rt || rt.gen !== gen || wantClose || !ws || ws.readyState !== 1) {
   387→        stopHbTimer();
   388→        return;
   389→      }
   390→      // 上一轮心跳没收到 ACK → 连接僵死，断开走重连（Resume 续传）
   391→      if (!q.hbOk) {
   392→        try {
   393→          ws.close();
   394→        } catch {
   395→          /* 忽略 */
   396→        }
   397→        return;
   398→      }
   399→      q.hbOk = false;
   400→      wsSend({ op: 1, d: q.lastSeq || null });
   401→    }, hbInterval);
   402→    if (q.sessionId && q.lastSeq) {
   403→      wsSend({ op: 6, d: { token: 'QQBot ' + token, session_id: q.sessionId, seq: q.lastSeq } });
   404→    } else {
   405→      wsSend({
   406→        op: 2,
   407→        d: {
   408→          token: 'QQBot ' + token,
   409→          intents: QQ_INTENTS,
   410→          shard: [0, 1],
   411→          properties: { $os: 'linux', $browser: 'im-gateway', $device: 'im-gateway' },
   412→        },
   413→      });
   414→    }
   415→    return;
   416→  }
   417→  if (op === 11) {
   418→    q.hbOk = true;
   419→    // 心跳 ACK 到达即代表会话已鉴权（Resume 后平台可能不发 RESUMED——心跳在线就是真在线）
   420→    if (q.state === 'connecting') q.state = 'ready';
   421→    return;
   422→  }
   423→  if (op === 7) {
   424→    // 服务端要求重连：立即断开走 Resume
   425→    console.log('[im-gateway] QQ WS op7 reconnect 指令');
   426→    try {
   427→      ws?.close();
   428→    } catch {
   429→      /* 忽略 */
   430→    }
   431→    return;
   432→  }
   433→  if (op === 9) {
   434→    // Identify/Resume 参数错（会话失效等）：重置会话重新鉴权
   435→    q.sessionId = '';
   436→    q.state = 'connecting';
   437→    q.lastError = 'QQ 平台返回会话无效（op9），重新鉴权中';
   438→    console.log('[im-gateway] QQ WS op9 invalid session:', JSON.stringify(f.d || null).slice(0, 120));
   439→    try {
   440→      ws?.close();
   441→    } catch {
   442→      /* 忽略 */
   443→    }
   444→    return;
   445→  }
   446→  if (op === 0) {
   447→    const t = String(f.t || '');
   448→    if (t === 'READY') {
   449→      q.state = 'ready';
   450→      q.sessionId = String(f.d?.session_id || '');
   451→      q.botName = String(f.d?.user?.username || f.d?.user?.nickname || '');
   452→      q.lastError = '';
   453→      reconnectDelay = 2000;
   454→      console.log('[im-gateway] QQ WS READY:', q.botName, 'session', q.sessionId.slice(0, 8));
   455→      return;
   456→    }
   457→    if (t === 'RESUMED') {
   458→      q.state = 'ready';
   459→      q.lastError = '';
   460→      reconnectDelay = 2000;
   461→      console.log('[im-gateway] QQ WS RESUMED: session', q.sessionId.slice(0, 8), 'seq', q.lastSeq);
   462→      return;
   463→    }
   464→    if (t === 'C2C_MESSAGE_CREATE' || t === 'GROUP_AT_MESSAGE_CREATE') {
   465→      q.lastEventAt = Date.now();
   466→      q.events += 1;
   467→      const d = f.d || {};
   468→      console.log('[im-gateway] QQ event:', t, 'id=', String(d.id || '').slice(0, 18));
   469→      const key = t + ':' + String(d.id || '');
   470→      const seenAt = qqSeen.get(key);
   471→      qqSeen.set(key, Date.now());
   472→      pruneSeen();
   473→      if (seenAt) return; // 平台重推的同一条消息：不重复回复
   474→      // 消息代答复用云端核心（人设/表情包/语音条/历史回写 Supabase 全在内）；cfg 走 60s 缓存
   475→      try {
   476→        const cfg = await rt.core.loadConfig(false);
   477→        const handled = await rt.core.handleQqEvent(cfg, t, d);
   478→        console.log('[im-gateway] QQ 处理', handled ? '已回复' : '跳过（无内容）');
   479→      } catch (e) {
   480→        q.lastError = String((e as Error)?.message || e).slice(0, 160);
   481→        console.log('[im-gateway] QQ 处理失败:', q.lastError);
   482→      }
   483→      return;
   484→    }
   485→    return; // 其它事件（好友添加等）忽略
   486→  }

   545→async function wxLoop(gen: number): Promise<void> {
   546→  let errs = 0;
   547→  // v15 定时主动消息检查节流：60s 一查（窗口命中才发；云端核心 maybeQqProactive 自带
   548→  // 当天已发/月 4 条自限，网关与云端并发也只发一次——占号先写后发）
   549→  let lastProactiveAt = 0;
   550→  while (rt && rt.gen === gen) {
   551→    const wx = rt.wx;
   552→    try {
   553→      const n = (await rt.core.tick()) as number;
   554→      errs = 0;
   555→      wx.lastRunAt = Date.now();
   556→      wx.lastError = '';
   557→      if (n > 0) wx.replied += n;
   558→      await rt.core.writeHeartbeat({ source: 'im-gateway' });
   559→      if (Date.now() - lastProactiveAt > 60000 && typeof rt.core.maybeQqProactive === 'function') {
   560→        lastProactiveAt = Date.now();
   561→        try {
   562→          const cfg2 = await rt.core.loadConfig(false);
   563→          const which = await rt.core.maybeQqProactive(cfg2);
   564→          if (which) console.log('[im-gateway] 定时主动消息已发：', which);
   565→        } catch {
   566→          /* 主动消息失败不拖拽收发 */
   567→        }
   568→      }
   569→    } catch (e) {
   570→      const msg = String((e as Error)?.message || e);
   571→      wx.lastError = msg.slice(0, 160);
   572→      wx.lastRunAt = Date.now();
   573→      if (msg.indexOf('TOKEN_EXPIRED') >= 0) {
   574→        // 微信 Token 过期：等主人回手机 App 重新扫码（期间静默，5 分钟再探一次）
   575→        await sleep(300000);
   576→      } else if (msg.indexOf('未配置') >= 0 || msg.indexOf('运行包缺失') >= 0) {
   577→        await sleep(60000);
   578→      } else {
   579→        errs++;
   580→        await sleep(Math.min(5000 * errs, 30000));
   581→      }
   582→      continue;
   583→    }
   584→    // 正常周期：getupdates 服务端挂 ~35s；空闲（未接入微信）也 10s 一探，不热转
   585→    await sleep(10000);
   586→  }
   587→}

   589→// ---------------- v14 僵尸自愈看门狗 ----------------
   590→
   591→/** 实测根因（用户反馈「QQ 接入过一会会断开 / AI 不回消息 / 断连」）：Bun 的 fetch 在
   592→ *  中继长连接上偶发挂死——AbortSignal.timeout 不触发（Bun 已知坑），wxLoop 卡在 tick()
   593→ *  里、connectQq 卡在取凭证/网关地址里：进程活着、HTTP 状态页活着，但收发全部停摆
   594→ *  （状态页上 lastRunAt / reconnects 永远停在同一秒）。旧拉活机制全部失灵：keeper 只看
   595→ *  端口、页面只看 running 标志、云端被连接锁挡住。
   596→ *  看门狗：每 20s 检查一次活力——①微信循环 lastRunAt 冻结 >150s（正常 ≤60s 一轮；
   597→ *  TOKEN_EXPIRED 的 5 分钟静默重探除外）；②QQ 长时间停在 connecting/error 且重试计数
   598→ *  不再推进（健康重试每 ≤30s 必动）→ 判定僵尸 → process.exit(1)：im-keeper.sh 检测端口
   599→ *  掉线秒级拉起 + gateway-state.json 启动自恢复 → 全量自愈（比永远僵尸强得多）。 */
   600→let gwWatchTimer: ReturnType<typeof setInterval> | null = null;
   601→let gwQqSig = '';
   602→let gwQqSigAt = 0;
   603→
   604→function startGwWatchdog(gen: number): void {
   605→  if (gwWatchTimer) clearInterval(gwWatchTimer);
   606→  gwQqSig = '';
   607→  gwQqSigAt = Date.now();
   608→  gwWatchTimer = setInterval(() => {
   609→    if (!rt || rt.gen !== gen) {
   610→      if (gwWatchTimer) clearInterval(gwWatchTimer);
   611→      return;
   612→    }
   613→    // ① 微信循环冻结（含启动宽限 30s：lastRunAt=0 时不判）
   614→    const wxStaleMs = Date.now() - rt.wx.lastRunAt;
   615→    const tokenWait = /TOKEN_EXPIRED|Token 已过期/.test(rt.wx.lastError);
   616→    if (rt.wx.lastRunAt > 0 && wxStaleMs > (tokenWait ? 400_000 : 150_000)) {
   617→      console.log('[im-gateway] 僵尸自愈：微信循环冻结 ' + Math.round(wxStaleMs / 1000) + 's → 重启进程（keeper 拉起 + 自恢复）');
   618→      process.exit(1);
   619→    }
   620→    // ② QQ 卡在 connecting/error 且不再推进（健康重试每 ≤30s 必动 reconnects/lastError）
   621→    const qqSig = [rt.qq.state, rt.qq.reconnects, rt.qq.events, rt.qq.lastError, rt.qq.lastSeq].join('|');
   622→    if (qqSig !== gwQqSig) {
   623→      gwQqSig = qqSig;
   624→      gwQqSigAt = Date.now();
   625→    } else if (
   626→      (rt.qq.state === 'connecting' || rt.qq.state === 'error') &&
   627→      Date.now() - gwQqSigAt > 150_000
   628→    ) {
   629→      console.log('[im-gateway] 僵尸自愈：QQ 连接停摆 ' + rt.qq.state + ' 超过 150s → 重启进程（keeper 拉起 + 自恢复）');
   630→      process.exit(1);
   631→    }
   632→  }, 20_000);
   633→}

   698→/** 探测云端函数代际：ping 动作（所有版本都无害）带 ver>=8 才支持 read-file 中继 */
   699→async function probeRelayCapability(relayUrl: string, relaySecret: string): Promise<{ ok: boolean; msg: string }> {
   700→  try {
   701→    const r = await fetch(relayUrl + '?action=ping', {
   702→      method: 'POST',
   703→      headers: { 'Content-Type': 'application/json', 'X-Task-Secret': relaySecret },
   704→      body: '{}',
   705→      signal: AbortSignal.timeout(15_000),
   706→    });
   707→    const j = (await r.json().catch(() => null)) as { ok?: boolean; ver?: number } | null;
   708→    if (r.status === 401) return { ok: false, msg: '任务密钥不匹配：请回手机 App 设置 → 云端助手 重新点「一键部署」升级函数' };
   709→    if (!j || !j.ok) return { ok: false, msg: '云端函数不可达或未部署（HTTP ' + r.status + '）——请确认已完成一键部署' };
   710→    if ((Number(j.ver) || 0) < 8) return { ok: false, msg: '云端函数是旧版（不带中继读能力）——请回手机 App 设置 → 云端助手 再点一次「一键部署」' };
   711→    return { ok: true, msg: '' };
   712→  } catch (e) {
   713→    return { ok: false, msg: '连不上云端函数：' + String((e as Error)?.message || e).slice(0, 80) };
   714→  }

   717→async function startGateway(supaUrl: string, supaKey: string, ref: string, relayUrl: string, relaySecret: string): Promise<{ ok: boolean; msg: string }> {
   718→  const url = String(supaUrl || '').replace(/\/+$/, '');
   719→  const key = String(supaKey || '');
   720→  const relay = String(relayUrl || '').replace(/\/+$/, '');
   721→  const rsecret = String(relaySecret || '');
   722→  if (!/^https:\/\/[a-z0-9-]+\.supabase\.co$/i.test(url)) return { ok: false, msg: 'supaUrl 不是有效的 Supabase 项目地址' };
   723→  if (rt && rt.ref === ref) return { ok: true, msg: '网关已在运行' };
   724→
   725→  // v8 通道决策：优先中继（任务密钥随部署注入，永远有效；用户 Storage key 可能被轮换全拒），
   726→  // 中继不可用时兑底直连密钥；两样都没有才报错。
   727→  let mode: 'relay' | 'direct' = 'relay';
   728→  if (rsecret && relay) {
   729→    const probe = await probeRelayCapability(relay, rsecret);
   730→    if (!probe.ok) {
   731→      if (key.length >= 20) {
   732→        mode = 'direct'; // 有直连密钥：降级直连，网关仍能跑
   733→        console.log('[im-gateway] 中继不可用，降级直连：', probe.msg);
   734→      } else {
   735→        return { ok: false, msg: '网关无法读云端配置：' + probe.msg };
   736→      }
   737→    }
   738→  } else if (key.length >= 20) {
   739→    mode = 'direct';
   740→  } else {
   741→    return { ok: false, msg: '缺少云端凭证（任务密钥/项目密钥都没有）：请回手机 App 重新一键部署' };
   742→  }
   743→
   744→  // 换连接：先停旧会话再立新
   745→  if (rt) await stopGateway();
   746→
   747→  // 中继模式落盘（自恢复用；直连模式不落盘——密钥不该进磁盘）
   748→  if (mode === 'relay') {
   749→    writePersistedState({ v: 1, supaUrl: url, ref: ref || url.replace(/^https:\/\//, '').split('.')[0], relayUrl: relay, relaySecret: rsecret, savedAt: Date.now() });
   750→  } else {
   751→    clearPersistedState();
   752→  }
   753→
   754→  const core = await materializeCore(url, mode === 'relay' ? '' : key, mode === 'relay' ? relay : '', mode === 'relay' ? rsecret : '');
   755→  // 启动自检：真实读一次运行包（密钥错/包缺失立刻报可读原因，不静默空转）
   756→  try {
   757→    await core.loadConfig(true);
   758→  } catch (e) {
   759→    const msg = String((e as Error)?.message || e);
   760→    clearPersistedState();
   761→    return { ok: false, msg: '读取云端运行包失败：' + msg.slice(0, 120) };
   762→  }
   763→  const gen = rt ? rt.gen + 1 : 1;
   764→  rt = {
   765→    gen,
   766→    ref: ref || url.replace(/^https:\/\//, '').split('.')[0],
   767→    startedAt: Date.now(),
   768→    mode,
   769→    core,
   770→    qq: { state: 'connecting', sessionId: '', botName: '', lastSeq: 0, lastEventAt: 0, events: 0, reconnects: 0, lastError: '', hbOk: false },
   771→    wx: { lastRunAt: 0, lastError: '', replied: 0 },
   772→  };
   773→  // qqSeen 跨连接保留无意义：清掉
   774→  qqSeen.clear();
   775→  wantClose = false;
   776→  reconnectDelay = 2000;
   777→  void wxLoop(gen);
   778→  void connectQq(gen);
   779→  // v14 僵尸自愈看门狗（fetch 挂死/循环停摆 → process.exit → keeper 拉起 + 自恢复）
   780→  startGwWatchdog(gen);
   781→  // v9：QQ 连接锁续期（每 15s）——QQ 在线期间云端函数 cron-cycle 看到新鲜锁就不另开
   782→  // 第二条连接（本网关是“本站在线时”的 QQ 持有方；本网关不在线时锁自然过期，云端接手）。
   783→  // v14：只有【真在线（ready）】才占锁——连接挂了（connecting/error）不再刷锁：旧版挂了
   784→  // 也继续刷，40s 内云端骨干永远接不了手 → 网关进程活着但连接死了 = 真机 QQ 全员掉线
   785→  // （「QQ 接入过一会会断开」的另一半根因）。现在连接一挂锁就过期，云端 cron-cycle
   786→  // 接手收发，网关自愈重连成功后（拿回 ready）重新占锁、云端下一轮让位。
   787→  if (qqLockTimer) clearInterval(qqLockTimer);
   788→  qqLockTimer = setInterval(() => {
   789→    if (!rt || rt.gen !== gen) {
   790→      if (qqLockTimer) clearInterval(qqLockTimer);
   791→      return;
   792→    }
   793→    if (rt.qq.state !== 'ready') return; // 没真在线不占锁（disabled 没人需要；connecting/error 让云端骨干接手）
   794→    void rt.core.refreshQqWsLock('gw').catch(() => undefined);
   795→  }, 15000);
   796→  // v12 凭证看门狗（每 30s）：QQ 在线期间重读运行包——手机 App「断开连接」后凭证段
   797→  // 被移除时拆长连接（真机 QQ 上 TA 下线）；网络读失败不打扰在线连接。
   798→  if (qqCfgTimer) clearInterval(qqCfgTimer);
   799→  qqCfgTimer = setInterval(() => {
   800→    if (!rt || rt.gen !== gen) {
   801→      if (qqCfgTimer) clearInterval(qqCfgTimer);
   802→      return;
   803→    }
   804→    if (rt.qq.state === 'disabled') return; // disabled 分支自己的 30s 重查已覆盖
   805→    void rt.core
   806→      .loadConfig(true)
   807→      .then((cfg: { qq?: { appId?: string } } | null) => {
   808→        if (!rt || rt.gen !== gen) return;
   809→        if (!cfg || !cfg.qq || !cfg.qq.appId) disableQqChannel(gen);
   810→      })
   811→      .catch(() => undefined);
   812→  }, 30000);
   813→  return { ok: true, msg: '网关已启动：QQ WebSocket 长连接 + 微信自动回复' + (mode === 'relay' ? '（云端函数中继通道）' : '') };
   814→}

   816→async function stopGateway(): Promise<{ ok: boolean; msg: string }> {
   817→  if (!rt) {
   818→    clearPersistedState();
   819→    return { ok: true, msg: '网关本就未在运行' };
   820→  }
   821→  const cur = rt;
   822→  rt = null;
   823→  stopQq();
   824→  if (gwWatchTimer) {
   825→    clearInterval(gwWatchTimer);
   826→    gwWatchTimer = null;
   827→  }
   828→  clearPersistedState(); // v8：自恢复状态一并清掉（重启后不再自动拉起）
   829→  cur.gen = -1; // 让微信循环自然退出（下一轮 while 判定失败）
   830→  return { ok: true, msg: '网关已停止（QQ 长连接已断开、微信自动回复已停）' };
   831→}
