   253→        signal: AbortSignal.timeout(75000),

   556→async function synthesizeQqVoice(cfg, text) {

  1313→      return jr({ ok: validCount > 0 && uploaded.length === validCount, uploaded: uploaded, error: lastUpErr, via: 'edge-function' });
  1314→    }
  1315→
  1316→    // ---- read-file（v8 内置网关中继读）：用户 Supabase 项目密钥拿不到/被拒时（代写路径部署后
  1317→    //    serviceKey 为空，或 JWT 轮换后旧 key 全拒），网关的配置/历史/游标读改走这里——函数用
  1318→    //    平台注入凭证读 Storage。鉴权同 upload-file（任务密钥优先，兑底比对桶里 config.json 的
  1319→    //    secret）；放在 loadConfig 之前同理（config 缺失时不能先抛「运行包缺失」）。
  1320→    if (action === 'read-file') {
  1321→      let cfgSecret = '';
  1322→      try {
  1323→        const c0 = await supaGetJson('config.json');
  1324→        cfgSecret = c0 && c0.secret ? String(c0.secret) : '';
  1325→      } catch (e) { /* 读取失败按没有处理 */ }
  1326→      const authed = !!(secret && (TASK_SECRET_ENV === secret || (cfgSecret && cfgSecret === secret)));
  1327→      if (!authed) return jr({ ok: false, error: 'bad secret' }, 401);
  1328→      let rb: { path?: unknown } = {};
  1329→      try { rb = (await req.json().catch(() => ({}))) as { path?: unknown }; } catch (e) { /* 空 body */ }
  1330→      const p = String(rb.path || '').replace(/[^a-zA-Z0-9._/-]/g, '');
  1331→      if (!p || p.length > 128) return jr({ ok: false, error: 'bad path' }, 400);
  1332→      const obj = await supaGetJson(p);
  1333→      if (obj === null || obj === undefined) return jr({ ok: false, error: 'not found' }, 404);
  1334→      return jr({ ok: true, relay: true, ver: 8, data: obj });
  1335→    }
  1336→
  1337→    const cfg = await loadConfig(false);
  1338→    // 任务密钥：桶里 config.secret 或部署时注入的 WX_TASK_SECRET 任一命中即可
  1339→    if (!(secret && (secret === (cfg && cfg.secret) || (TASK_SECRET_ENV && secret === TASK_SECRET_ENV)))) {
  1340→      return jr({ ok: false, error: 'bad secret' }, 401);
  1341→    }
  1342→
  1343→    // ---- 管理 action ----
  1344→    // v5：setup-cron / stop-cron 已废弃（定时任务改由手机 App 设置里的「云端轮询」开关
  1345→    // 走服务端 Management API SQL 建立/清除，不再依赖函数里的 npm:postgres）。兼容老客户端的调用
  1346→    if (action === 'setup-cron' || action === 'stop-cron') {
  1347→      return jr({ ok: true, note: '定时任务请用手机 App 设置 → 云端助手 的「云端轮询」开关管理（v6 起）；本函数已改为被动 Webhook + 单轮收发' });
  1348→    }
  1349→    // 云端验证 QQ 凭证（绕开本机网络限制：从这台云服务器的出口连 QQ 平台）
  1350→    if (action === 'verify-qq') {
  1351→      let body = { appId: '', appSecret: '' };
  1352→      try { body = await req.json(); } catch (e) { /* 空 body */ }
  1353→      const r = await verifyQqCreds(body.appId, body.appSecret);
  1354→      return jr(r);
  1355→    }
  1356→    if (action === 'ping') {
  1357→      await writeHeartbeat({ ping: true });
  1358→      // ver：内置网关探测函数代际用（read-file 中继读写 = ver>=8；旧版函数 ping 无 ver 字段）
  1359→      return jr({ ok: true, source: 'deno', ver: 9 });
  1360→    }
  1361→
  1362→    // ---- v9 cron-cycle（Supabase 定时任务每分钟打一次）：微信单轮收发 + QQ WebSocket
  1363→    // 会话保持到预算——云端 7×24 骨干，不依赖本站在线。会话断点存 state/qq-ws-session.json，
  1364→    // 下轮 Resume 续传（平台补发断线期间的事件）；本站网关在线时（qq-ws-lock）自动让位。 ----
  1365→    if (action === 'cron-cycle') {
  1366→      const t0 = Date.now();
  1367→      let handled = 0;
  1368→      let lastError = '';
  1369→      try {
  1370→        handled += await tick();
  1371→      } catch (e) {
  1372→        lastError = String(e && e.message);
  1373→      }
  1374→      let qq = { skipped: 'no-qq' };
  1375→      // 微信轮询这轮还没把预算吃光才开 QQ 保持（消息多/LLM 慢时优先微信）
  1376→      if (Date.now() - t0 < 100000) {
  1377→        try {
  1378→          qq = await qqWsHold(cfg, 135000 - (Date.now() - t0));
  1379→        } catch (e) {
  1380→          qq = { error: String(e && e.message).slice(0, 120) };
  1381→        }
  1382→      }
  1383→      await writeHeartbeat({ handled: handled, lastError: lastError, source: 'cron-cycle' });
  1384→      return jr({ ok: !lastError, handled: handled, lastError: lastError, wxMsgError: WX_MSG_ERR || '', qq: qq, ranMs: Date.now() - t0 });
  1385→    }
  1386→
  1387→    // ---- 默认：微信单轮收发（手动「云端测试一次」触发；一次 getupdates 服务端挂
  1388→    // ~35s，零 CPU 热转——旧版 100s 热循环轮询是 546 WORKER_RESOURCE_LIMIT 的根因；
  1389→    // 定时任务已由 cron-cycle 承担，本分支只做单轮验证/补收） ----
  1390→    const t0 = Date.now();
  1391→    let handled = 0;
  1392→    let lastError = '';
  1393→    try {
  1394→      handled += await tick();
  1395→    } catch (e) {
  1396→      lastError = String(e && e.message);
  1397→    }
  1398→    await writeHeartbeat({ handled: handled, lastError: lastError });
  1399→    return jr({ ok: !lastError, handled: handled, lastError: lastError, wxMsgError: WX_MSG_ERR || '', ranMs: Date.now() - t0 });
  1400→  } catch (e) {
  1401→    return jr({ ok: false, error: String(e && e.message).slice(0, 200) }, 500);
  1402→  }
