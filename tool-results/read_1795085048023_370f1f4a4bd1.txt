   124→async function ilink<T = Record<string, unknown>>(
   125→  path: string,
   126→  opts: { botToken?: string; body?: unknown; method?: 'GET' | 'POST'; host?: string; timeoutMs?: number } = {},
   127→): Promise<T> {
   128→  const res = await fetch('/api/weixin-ilink', {
   129→    method: 'POST',
   130→    headers: { 'Content-Type': 'application/json' },
   131→    body: JSON.stringify({
   132→      path,
   133→      method: opts.method ?? 'POST',
   134→      botToken: opts.botToken,
   135→      body: opts.body ?? {},
   136→      ...(opts.host ? { host: opts.host } : {}),
   137→    }),
   138→    // 状态长轮询与 getupdates 服务端挂 ~35s：客户端窗留足 50s，其余 40s
   139→    signal: AbortSignal.timeout(opts.timeoutMs ?? 40_000),
   140→  });
   141→  const text = await res.text();
   142→  let j: unknown;
   143→  try {
   144→    j = JSON.parse(text);
   145→  } catch {
   146→    throw new Error(`iLink 响应异常（${res.status}）：${text.slice(0, 120)}`);
   147→  }
   148→  if (!res.ok) {
   149→    const rec = j as { error?: string; message?: string };
   150→    throw new Error(rec.message || rec.error || `iLink 请求失败（${res.status}）`);
   151→  }
   152→  return j as T;
   153→}

   258→const WX_CLASSIC_EMOJI_SET = new Set(BUILTIN_WX_STICKERS.map((s) => s.meaning));
   259→
   260→/** 纯表情净化后的兜底颜文字（回复整条只剩大表情标记时，不至于变空消息） */
   261→const WX_EMOJI_FALLBACKS = ['(¬‿¬)', '(๑•̀ㅂ•́)و✧', '٩(˃ᴗ˂)۶', '(っ˘ω˘ς)'];
   262→
   263→/**
   264→ * 真机微信出站净化：大表情标记 → 经典表情代码（能发的最大化），发不了的剥除。
   265→ * - [表情包:wxb-调皮] / [表情包:wwb-调皮] / [表情:调皮] 等标记：去 ID 前缀后名字在经典
   266→ *   表情清单内 → 转 [调皮]（真机微信直接显示成真表情，AI 等于真会发表情了）；不在清单
   267→ *   （用户收藏大表情）→ 剥除（文字通道发不了图，留着只会是奇怪文字）
   268→ * - [图片:xx] / [照片:xx] / [语音:xx] 等媒体标记 → 剥除
   269→ * - 已是 [微笑] 形态的经典代码零改动；净空后拿颜文字兑底
   270→ * 根因（用户实拍）：本机历史里站内微信的 [表情包:ID] 消息教会了 LLM 这个格式，而真机
   271→ * 通道发不出去——提示词拦不住，必须在发送边界确定性转换。
   272→ */
   273→export function sanitizeWxOutText(text: string): string {
   274→  const tagRe = /[\[【]\s*(?:发送了|发出了)?\s*(?:表情包|表情|图片|照片|语音|贴纸|Sticker)\s*[:：]\s*([^\]】]{1,24})\s*[\]】]/g;
   275→  let out = text.replace(tagRe, (_all, rawName: string) => {
   276→    const name = String(rawName)
   277→      .trim()
   278→      .replace(/^(?:wwb|wxb|qqb|wx|qq|mt|wb)-/i, '')
   279→      .trim();
   280→    return WX_CLASSIC_EMOJI_SET.has(name) ? `[${name}]` : '';
   281→  });
   282→  out = out
   283→    .replace(/\n{3,}/g, '\n\n')
   284→    .replace(/^[\s\n]+|[\s\n]+$/g, '')
   285→    .replace(/^[，,。.\s]+/, '');
   286→  if (out) return out;
   287→  return WX_EMOJI_FALLBACKS[Math.floor(Math.random() * WX_EMOJI_FALLBACKS.length)];
   288→}

   355→/** 处理一条来消息：落线程 → 打字状态 → 人设回复 → 发回微信 + 落线程 */
   356→async function handleIncoming(st: WxConnectStore, login: WxIlinkLogin, msg: ILinkMessage): Promise<void> {
   357→  const textItem = (msg.item_list ?? []).find((i) => i.type === 1);
   358→  const text = (textItem?.text_item?.text ?? '').trim();
   359→  if (!text || !msg.from_user_id || !msg.context_token) return;
   360→  if (msg.from_user_id === login.botId) return; // 自己发出的（多端同收）
   361→
   362→  const entry: WxIlinkMsg = {
   363→    id: `${msg.message_id ?? Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   364→    fromUserId: msg.from_user_id,
   365→    text: text.slice(0, 2000),
   366→    ts: Date.now(),
   367→  };
   368→  st.pushMsg(entry);
   369→  // 对方（通常是你自己）说的 → 线程里记为「我方(me)」消息
   370→  appendThreadMsg(login.boundContactId, { id: `wxr-${entry.id}`, role: 'me', content: text.slice(0, 2000), time: Date.now() });
   371→  lastCtx = { fromUserId: msg.from_user_id, contextToken: msg.context_token };
   372→
   373→  // 「正在输入」：getconfig 拿 ticket，回复期间每 4.5s 续一次（host 跟随 login.baseUrl）
   374→  const api = { host: hostOf(login.baseUrl) };
   375→  let typingTicket = '';
   376→  let typingTimer: ReturnType<typeof setInterval> | null = null;
   377→  try {
   378→    const cfg = await ilink<{ ret?: number; typing_ticket?: string }>('/ilink/bot/getconfig', {
   379→      botToken: login.botToken,
   380→      host: api.host,
   381→      body: { ilink_user_id: msg.from_user_id, context_token: msg.context_token, base_info: BASE_INFO },
   382→    });
   383→    typingTicket = cfg.typing_ticket ?? '';
   384→  } catch {
   385→    /* 打字状态可有可无 */
   386→  }
   387→  const stopTyping = () => {
   388→    if (typingTimer) clearInterval(typingTimer);
   389→    typingTimer = null;
   390→    if (typingTicket) {
   391→      void ilink('/ilink/bot/sendtyping', {
   392→        botToken: login.botToken,
   393→        host: api.host,
   394→        body: { ilink_user_id: msg.from_user_id, typing_ticket: typingTicket, status: 2, base_info: BASE_INFO },
   395→      }).catch(() => undefined);
   396→    }
   397→  };
   398→  if (typingTicket) {
   399→    const beat = () =>
   400→      void ilink('/ilink/bot/sendtyping', {
   401→        botToken: login.botToken,
   402→        host: api.host,
   403→        body: { ilink_user_id: msg.from_user_id, typing_ticket: typingTicket, status: 1, base_info: BASE_INFO },
   404→      }).catch(() => undefined);
   405→    beat();
   406→    typingTimer = setInterval(beat, 4500);
   407→  }
   408→
   409→  try {
   410→    // 组 prompt：人设 system + 绑定联系人线程的最近历史（真机/网页两侧聊天都在线程里，天然互通）
   411→    const thread = loadMsgs(login.boundContactId).filter((m) => !m.kind || m.kind === 'text').slice(-14);
   412→    const history = thread.slice(0, -1).map((m) => ({ role: m.role === 'me' ? ('user' as const) : ('assistant' as const), content: m.content.slice(0, 800) }));
   413→    const contextText = thread.map((m) => `${m.role === 'me' ? '我' : 'TA'}: ${m.content}`).join('\n').slice(-1500);
   414→    const system = await buildSystemPrompt(login, contextText);
   415→    const reply = sanitizeWxOutText((await callLlm([...history, { role: 'user', content: text.slice(0, 2000) }])).trim());
   416→    if (!reply) throw new Error('生成了空回复');
   417→
   418→    await ilink('/ilink/bot/sendmessage', {
   419→      botToken: login.botToken,
   420→      host: api.host,
   421→      body: {
   422→        msg: {
   423→          from_user_id: '',
   424→          to_user_id: msg.from_user_id,
   425→          client_id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   426→          message_type: 2,
   427→          message_state: 2,
   428→          context_token: msg.context_token,
   429→          item_list: [{ type: 1, text_item: { text: reply.slice(0, 2000) } }],
   430→        },
   431→        base_info: BASE_INFO,
   432→      },
   433→    });
   434→    st.patchMsg(entry.id, { reply: reply.slice(0, 2000) });
   435→    // 角色说的 → 线程里记为「对方(peer)」，并标记为我们转发的（别再往微信发一遍）
   436→    const peerId = `wxr-${entry.id}-r`;
   437→    knownPeerIds.add(peerId);
   438→    appendThreadMsg(login.boundContactId, { id: peerId, role: 'peer', content: reply.slice(0, 2000), time: Date.now() });
   439→  } catch (e) {
   440→    st.patchMsg(entry.id, { replyErr: e instanceof Error ? e.message : String(e) });
   441→  } finally {
   442→    stopTyping();
   443→  }
   444→}

   448→/** 已知「对方(peer)」消息 id 集合：我们自己落的不转发；boot 时把存量全部视为已知（不发历史） */
   449→const knownPeerIds = new Set<string>();

   451→async function forwardThreadWatcher(login: WxIlinkLogin): Promise<void> {
   452→  try {
   453→    const list = loadMsgs(login.boundContactId);
   454→    const fresh = list.filter((m) => m.role === 'peer' && (!m.kind || m.kind === 'text') && !knownPeerIds.has(m.id));
   455→    for (const m of fresh) {
   456→      knownPeerIds.add(m.id);
   457→      if (!lastCtx) continue; // 还没有任何微信来消息，没有 context_token 可用
   458→      await ilink('/ilink/bot/sendmessage', {
   459→        botToken: login.botToken,
   460→        host: hostOf(login.baseUrl),
   461→        body: {
   462→          msg: {
   463→            from_user_id: '',
   464→            to_user_id: lastCtx.fromUserId,
   465→            client_id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   466→            message_type: 2,
   467→            message_state: 2,
   468→            context_token: lastCtx.contextToken,
   469→            item_list: [{ type: 1, text_item: { text: sanitizeWxOutText(m.content).slice(0, 2000) } }],
   470→          },
   471→          base_info: BASE_INFO,
   472→        },
   473→      }).catch(() => undefined);
   474→    }
   475→  } catch {
   476→    /* 转发尽力而为 */
   477→  }
   478→}

   480→// ---------------- 本机轮询主循环 ----------------
   481→
   482→async function runLoop(gen: number): Promise<void> {
   483→  const st = useWxConnect.getState();
   484→  st.setLoop('running', '');
   485→  while (gen === loopGen) {
   486→    const { login, autoReply } = useWxConnect.getState();
   487→    if (!login || !autoReply) break;
   488→    // v8 双收双回防抖（全页面生效——原来只有接入页让位）：内置网关在跑时本机轮询让位，
   489→    // 同一 botToken 两个轮询器会同收同回（重复回复）。让位期间仍跑网页侧新回复转发
   490→    // （网站里打的字仍要送到真机微信），15s 后再看网关状态；网关停了自动收回本机轮询。
   491→    try {
   492→      const gw = await fetchImGatewayStatus();
   493→      if (gw?.running) {
   494→        await forwardThreadWatcher(login);
   495→        await sleep(15000);
   496→        continue;
   497→      }
   498→    } catch {
   499→      /* 状态探不到（网关服务未起）不影响本机轮询 */
   500→    }
   501→    try {
   502→      const d = await ilink<{ ret?: number; error_code?: number; msgs?: ILinkMessage[]; get_updates_buf?: string }>(
   503→        '/ilink/bot/getupdates',
   504→        // host 跟随 confirmed 回包的 baseurl（换域后继续可用）；长轮询窗 50s
   505→        { botToken: login.botToken, host: hostOf(login.baseUrl), timeoutMs: 50_000, body: { get_updates_buf: updatesBuf, base_info: BASE_INFO } },
   506→      );
   507→      if (gen !== loopGen) break;
   508→      if (d.error_code === -14) {
   509→        useWxConnect.getState().setLoop('error', 'Token 已过期，请回到微信重新扫码接入');
   510→        return;
   511→      }
   512→      if (typeof d.get_updates_buf === 'string' && d.get_updates_buf) {
   513→        updatesBuf = d.get_updates_buf;
   514→        // v5：游标落盘（页面刷新后不重收积压——重复回复风暴的根因）
   515→        try {
   516→          kvSet(BUF_KEY, updatesBuf);
   517→        } catch {
   518→          /* 忽略 */
   519→        }
   520→      }
   521→      for (const m of d.msgs ?? []) {
   522→        if (gen !== loopGen) break;
   523→        await handleIncoming(useWxConnect.getState(), login, m).catch(() => undefined);
   524→      }
   525→      // 网页侧新回复转发（每轮一次）
   526→      await forwardThreadWatcher(login);
   527→      useWxConnect.setState({ lastPollAt: Date.now() });
   528→      if (useWxConnect.getState().loopStatus === 'error') {
   529→        useWxConnect.getState().setLoop('running', '');
   530→      }
   531→    } catch (e) {
   532→      if (gen !== loopGen) return;
   533→      useWxConnect.getState().setLoop('error', e instanceof Error ? e.message : String(e));
   534→      await sleep(5000);
   535→    }
   536→  }
   537→  if (gen === loopGen) useWxConnect.getState().setLoop('off', '');
   538→}

   542→interface WxConnectStore {
   543→  ready: boolean;
   544→  login: WxIlinkLogin | null;
   545→  autoReply: boolean;
   546→  loopStatus: 'off' | 'running' | 'error';
   547→  loopError: string;
   548→  lastPollAt: number | null;
   549→  msgs: WxIlinkMsg[];
   550→  qr: WxIlinkQr | null;
   551→  qrLoading: boolean;
   552→  testing: boolean;
   553→  lastTest: { ok: boolean; msg: string; at: number } | null;
   554→  /** 云端聊天记录拉取中（拉取云端代答的真机微信对话合进手机聊天界面） */
   555→  pulling: boolean;
   556→
   557→  boot: () => Promise<void>;
   558→  /** 选好绑定联系人后取二维码并开始官方协议状态机（长轮询 + PIN + 重定向 + 换码） */
   559→  startQr: (contactId: string, contactName: string) => Promise<void>;
   560→  /** 提交微信 PIN（need_verifycode 时 UI 输入 → 下一轮轮询带 verify_code） */
   561→  submitPin: (pin: string) => void;
   562→  cancelQr: () => void;
   563→  /** 换绑角色（不重新扫码，只改绑定的联系人） */
   564→  rebindContact: (contactId: string, contactName: string) => Promise<void>;
   565→  disconnect: () => Promise<void>;
   566→  /** 测试连接：getupdates 一次（有效 token 会快速返回，无效/过期会报错） */
   567→  test: () => Promise<boolean>;
   568→  setAutoReply: (v: boolean) => void;
   569→  pushMsg: (m: WxIlinkMsg) => void;
   570→  patchMsg: (id: string, patch: Partial<WxIlinkMsg>) => void;
   571→  clearMsgs: () => void;
   572→  /** 拉取云端微信聊天记录（测试云端/云端轮询代答的对话同步进手机聊天界面） */
   573→  pullHistory: () => Promise<void>;
   574→  setLoop: (s: 'off' | 'running' | 'error', err: string) => void;
   575→}
   576→
   577→/** 扫码状态机代数：start/stop/cancel 递增，旧循环自动退出 */
   578→let qrGen = 0;
   579→/** PIN 待提交的 resolver（need_verifycode 时 UI 输入后唤醒轮询循环） */
   580→let pinResolve: ((pin: string) => void) | null = null;
   581→
   582→export const useWxConnect = create<WxConnectStore>((set, get) => ({
   583→  ready: false,
   584→  login: null,
   585→  autoReply: false,
   586→  loopStatus: 'off',
   587→  loopError: '',
   588→  lastPollAt: null,
   589→  msgs: [],
   590→  qr: null,
   591→  qrLoading: false,
   592→  testing: false,
   593→  lastTest: null,
   594→  pulling: false,
   595→
   596→  boot: async () => {
   597→    if (wxConnectReady) return;
   598→    wxConnectReady = true;
   599→    try {
   600→      await ensureKvReady();
   601→      loadTokenHistory();
   602→      memLogin = await readEncJson<WxIlinkLogin>(LOGIN_KEY);
   603→      if (memLogin?.botToken) rememberToken(memLogin.botToken);
   604→      const msgs = await readEncJson<WxIlinkMsg[]>(MSGS_KEY).catch(() => null);
   605→      // v5：恢复游标（上次会话收过的不再重收——刷新页面后不会把积压重新回一遍）
   606→      try {
   607→        const savedBuf = kvGet<string>(BUF_KEY);
   608→        if (savedBuf) updatesBuf = savedBuf;
   609→      } catch {
   610→        /* 忽略 */
   611→      }
   612→      const auto = (() => {
   613→        try {
   614→          return kvGet<string>(AUTO_KEY) === '1';
   615→        } catch {
   616→          return false;
   617→        }
   618→      })();
   619→      const list = Array.isArray(msgs) ? msgs.slice(-MSGS_CAP) : [];
   620→      set({ login: memLogin, msgs: list, autoReply: auto && !!memLogin, ready: true });
   621→      if (memLogin) {
   622→        // 存量对方消息全部视为已知（不把历史回复重发到微信）
   623→        try {
   624→          for (const m of loadMsgs(memLogin.boundContactId)) {
   625→            if (m.role === 'peer') knownPeerIds.add(m.id);
   626→          }
   627→        } catch {
   628→          /* 忽略 */
   629→        }
   630→        if (auto) {
   631→          loopGen++;
   632→          void runLoop(loopGen);
   633→        }
   634→      }
   635→    } catch {
   636→      set({ ready: true });
   637→    }
   638→  },
   639→
   640→  /**
   641→   * 官方协议状态机（Tencent/openclaw-weixin docs/protocol_zh_CN.md）：
   642→   * ① POST get_bot_qrcode?bot_type=3，请求体 {local_token_list}（本机最近 10 个 bot token——
   643→   *   后端靠它认出环境亲缘，重绑不被当陌生环境风控）
   644→   * ② GET get_qrcode_status?qrcode=… 串行长轮询（服务端挂 ~35s；超时/网络错 = wait 续轮）。
   645→   *    状态：wait / scaned / confirmed（保存 bot_token+ilink_bot_id+baseurl+ilink_user_id）/
   646→   *    expired / 含 need_verifycode（手机显示数字 → 用户输入 → 下一轮带 verify_code，
   647→   *    再遇 need_verifycode = 输错重输）/ verify_code_blocked（错太多，换码重试 ≤3）/
   648→   *    scaned_but_redirect（后续轮询切 redirect_host）/ binded_redirect（已绑其它环境，终态引导）
   649→   * ③ 过期/锁死自动换码重试 3 次，超限提示稍后再试。
   650→   */
   651→  startQr: async (contactId, contactName) => {
   652→    if (get().qrLoading) return;
   653→    const myGen = ++qrGen;
   654→    set({ qrLoading: true, qr: null });
   655→    try {
   656→      await ensureKvReady();
   657→      // 换码重试外环（expired / verify_code_blocked ≤ 3 次）
   658→      for (let refresh = 0; refresh <= 3 && qrGen === myGen; refresh++) {
   659→        const d = await ilink<{ qrcode?: string; qrcode_img_content?: string; ret?: number }>(
   660→          '/ilink/bot/get_bot_qrcode?bot_type=3',
   661→          // local_token_list：本机 token 历史（协议：最多 10 个，后端认环境亲缘用）
   662→          { body: { local_token_list: tokenHistorySnapshot() } },
   663→        );
   664→        if (qrGen !== myGen) return;
   665→        if (!d.qrcode || !d.qrcode_img_content) throw new Error('iLink 未返回二维码');
   666→        // qrcode_img_content 是授权页 URL（不是图片）——本地生成二维码 data URL
   667→        const img = await QRCode.toDataURL(d.qrcode_img_content, { width: 320, margin: 2 });
   668→        set({ qr: { qrcode: d.qrcode, img, status: 'wait' }, qrLoading: false });
   669→
   670→        const qrcode = d.qrcode;
   671→        let statusHost: string | undefined; // scaned_but_redirect 后续切 redirect_host
   672→        let pin = ''; // 暂存 PIN：仅随「下一次」轮询上报（协议约定）
   673→        let wasPin = false; // 上一轮是否处于 PIN 输入（区分首次要求与输错重输）
   674→        interface QrStatusResp {
   675→          status?: string;
   676→          bot_token?: string;
   677→          ilink_bot_id?: string;
   678→          baseurl?: string;
   679→          ilink_user_id?: string;
   680→          redirect_host?: string;
   681→        }
   682→        // 串行长轮询内环
   683→        while (qrGen === myGen) {
   684→          const cur = get().qr;
   685→          if (!cur || cur.qrcode !== qrcode) return; // 被新码/取消取代
   686→          const qs = `qrcode=${encodeURIComponent(qrcode)}${pin ? `&verify_code=${encodeURIComponent(pin)}` : ''}`;
   687→          let s: QrStatusResp | null = null;
   688→          try {
   689→            s = await ilink<QrStatusResp>(`/ilink/bot/get_qrcode_status?${qs}`, {
   690→              method: 'GET',
   691→              body: {},
   692→              host: statusHost,
   693→              timeoutMs: 50_000,
   694→            });
   695→          } catch {
   696→            // 协议：单次失败/超时按 wait 续轮（短歇后重试）
   697→            await sleep(1000);
   698→            continue;
   699→          }
   700→          if (qrGen !== myGen) return;
   701→          const status = String(s?.status || 'wait');
   702→          // ① confirmed：保存凭证（含 baseurl / ilink_user_id）
   703→          if (status === 'confirmed' && s?.bot_token && s?.ilink_bot_id) {
   704→            const login: WxIlinkLogin = {
   705→              botToken: s.bot_token,
   706→              botId: s.ilink_bot_id,
   707→              ...(typeof s.baseurl === 'string' && s.baseurl ? { baseUrl: s.baseurl } : {}),
   708→              ...(typeof s.ilink_user_id === 'string' && s.ilink_user_id ? { ilinkUserId: s.ilink_user_id } : {}),
   709→              boundContactId: contactId,
   710→              boundContactName: contactName,
   711→              connectedAt: Date.now(),
   712→            };
   713→            memLogin = login;
   714→            await writeEncJson(LOGIN_KEY, login);
   715→            rememberToken(login.botToken);
   716→            personaCache = null;
   717→            lastCtx = null;
   718→            updatesBuf = '';
   719→            pinResolve = null;
   720→            set({ login, qr: { ...cur, status: 'confirmed', pinErr: undefined }, msgs: [], lastTest: null });
   721→            // 扫码接入成功后默认开启本机自动回复（用户可关；24h 在线建议部署云端）
   722→            get().setAutoReply(true);
   723→            // 已部署过云端则自动重推运行包（微信通道即刻生效，无需手动点同步）
   724→            cloudAutoSync('微信接入成功');
   725→            return;
   726→          }
   727→          // ② 需要输入 PIN（含 need_verifycode / confirmed_but_need_verifycode）
   728→          if (status.includes('need_verifycode')) {
   729→            set({
   730→              qr: {
   731→                ...cur,
   732→                status: 'need_verifycode',
   733→                pinErr: wasPin ? '数字不对，看看微信上显示的是多少，重新输一次' : undefined,
   734→              },
   735→            });
   736→            const typed = await new Promise<string>((resolve) => {
   737→              pinResolve = resolve;
   738→            });
   739→            if (qrGen !== myGen) return;
   740→            pin = typed.trim().replace(/\D/g, '').slice(0, 8);
   741→            wasPin = true;
   742→            continue; // 下一轮带 verify_code（返回 need_verifycode 依旧 = 输错）
   743→          }
   744→          // ③ 已扫码（带过 verify_code 时表示数字已接受 → 丢弃）
   745→          if (status === 'scaned') {
   746→            pin = '';
   747→            wasPin = false;
   748→            if (cur.status !== 'scaned') set({ qr: { ...cur, status: 'scaned', pinErr: undefined } });
   749→            continue;
   750→          }
   751→          // ④ 过期 / 错码锁死：换码重试（≤3），超限退出引导
   752→          if (status === 'expired' || status === 'verify_code_blocked') {
   753→            pin = '';
   754→            pinResolve = null;
   755→            if (refresh >= 3) {
   756→              set({
   757→                qr: {
   758→                  ...cur,
   759→                  status: 'expired',
   760→                  hint: status === 'verify_code_blocked' ? '数字错太多次被锁：稍后再试，或点重新获取' : undefined,
   761→                },
   762→              });
   763→              return;
   764→            }
   765→            break; // 外环换新码
   766→          }
   767→          // ⑤ 扫码后换域：后续轮询切 redirect_host
   768→          if (status === 'scaned_but_redirect') {
   769→            const rh = String(s?.redirect_host || '').trim();
   770→            if (rh) statusHost = rh.replace(/^https?:\/\//i, '').split('/')[0].toLowerCase();
   771→            if (cur.status !== 'scaned') set({ qr: { ...cur, status: 'scaned', pinErr: undefined } });
   772→            continue;
   773→          }
   774→          // ⑥ 已绑到其它环境（终态）：给解绑引导，停止轮询
   775→          if (status === 'binded_redirect') {
   776→            pinResolve = null;
   777→            set({
   778→              qr: {
   779→                ...cur,
   780→                status: 'binded_redirect',
   781→                hint: '这个微信号已连接到其它环境（比如之前连过别处的机器人）：先在那边断开/解绑，或换一个微信号扫码',
   782→              },
   783→            });
   784→            return;
   785→          }
   786→          // ⑦ wait：继续长轮询
   787→          if (cur.status !== 'wait') set({ qr: { ...cur, status: 'wait', pinErr: undefined } });
   788→        }
   789→      }

   939→/** 云端 wx-cloud/history.json 的条目（角色视角：user=真机微信上发消息的人、assistant=TA 的回复） */
   940→interface WxCloudHistItem {
   941→  role?: string;

   946→/**
   947→ * 拉取云端微信聊天记录：云端助手代答（测试云端/定时轮询代跑）收发的对话存在 Supabase
   948→ * Storage wx-cloud/history.json（最近 30 条）——拉回来按「真机微信上的人=me、TA=peer」合进
   949→ * 手机里与绑定角色的微信聊天界面（去重：确定性 id，拉过的不会重复入列）。
   950→ * 与「本机实时回复」互不干扰：本机路径的消息已直接落聊天界面；这里只补云端路径的记录
   951→ * （用户实测反馈：点「测试云端」后 TA 在真机微信里回复了，但网站聊天界面里看不到——
   952→ * 云端代答的对话只有云端有，拉一次即合进来看）。
   953→ * v12 单飞闸：手动按钮 / 接入页 / cloud-history-sync 自动轮询 / store 动作多路并发
   954→ * 调用时共享同一轮在飞结果——防并发双写本地线程（同 QQ 侧 pullQqCloudHistory）。
   955→ */
   956→let wxPullInFlight: Promise<{ ok: boolean; msg: string; added?: number }> | null = null;
   957→
   958→export function pullWxCloudHistory(): Promise<{ ok: boolean; msg: string; added?: number }> {
   959→  if (!wxPullInFlight) {
   960→    wxPullInFlight = doPullWxCloudHistory().finally(() => {
   961→      wxPullInFlight = null;
   962→    });
   963→  }
   964→  return wxPullInFlight;
   965→}
   966→
   967→async function doPullWxCloudHistory(): Promise<{ ok: boolean; msg: string; added?: number }> {
   968→  const cloud = wxCloudSnapshot();
   969→  const supa = cloud.supabase;
   970→  const login = memLogin;
   971→  if (!login) return { ok: false, msg: '还没接入真机微信' };
   972→  if (!login.boundContactId) return { ok: false, msg: '还没绑定角色：请先选择一个 AI 角色' };
   973→  // v8：读历史双通道——有 serviceKey 直连 Storage；没有（代写路径部署后为空、或 key 被项目
   974→  // 拒收）经云端函数 read-file 中继（与内置网关同一条通道）。原来 serviceKey 为空直接拦
   975→  // 「云端未就绪」——部署明明成功了却永远拉不到记录，用户实测就是这里。
   976→  if (!supa?.ref || (!supa.serviceKey && !supa.secret)) return { ok: false, msg: '云端未就绪：请先到 设置 → 云端助手 完成部署' };
   977→  try {
   978→    let j: { msgs?: WxCloudHistItem[] } | null = null;
   979→    if (supa.serviceKey) {
   980→      // 新式 sb_secret_ key 只能放 apikey 头（Bearer 会被当 JWT 报 400 Invalid Compact JWS）
   981→      const k = supa.serviceKey;
   982→      const isNewStyle = k.startsWith('sb_secret_') || k.startsWith('sb_publishable_');
   983→      const res = await fetch(`https://${supa.ref}.supabase.co/storage/v1/object/wx-cloud/history.json`, {
   984→        headers: isNewStyle ? { apikey: k } : { apikey: k, Authorization: `Bearer ${k}` },
   985→        signal: AbortSignal.timeout(15_000),
   986→      });
   987→      if (!res.ok) return { ok: false, msg: '云端还没有微信聊天记录（真机微信上和 TA 聊过之后才有）' };
   988→      j = (await res.json()) as { msgs?: WxCloudHistItem[] };
   989→    } else {
   990→      // 动态 import 防模块环（supabase.ts → run-package.ts → 本文件）
   991→      const { readBucketJsonViaRelay } = await import('./wx-cloud/supabase');
   992→      j = await readBucketJsonViaRelay<{ msgs?: WxCloudHistItem[] }>(supa.ref, 'history.json');
   993→      if (!j) {
   994→        return {
   995→          ok: false,
   996→          msg: '云端还没有微信聊天记录（真机微信上和 TA 聊过之后才有）。若 TA 在真机上一直不回：回 设置 → 云端助手 重新点一次「一键部署」（升级云端函数）再看',
   997→        };
   998→      }
   999→    }
  1000→    const list = Array.isArray(j.msgs) ? j.msgs : [];
  1001→    const cur = loadMsgs(login.boundContactId);
  1002→    const seen = new Set(cur.map((m) => m.id));
  1003→    const add: WxMsg[] = [];
  1004→    for (const m of list) {
  1005→      if (!m || typeof m.content !== 'string' || !m.content.trim()) continue;
  1006→      const mine = m.role === 'user'; // 云端视角：user=真机微信上发消息的人（=机主）→ 站内 me
  1007→      const ts = Number(m.ts) || Date.now();
  1008→      // 确定性 id：ts+角色+内容散列——同一云端记录重复拉取不重复入列（head 截断不撞号）
  1009→      let h = 0;
  1010→      const c = m.content.slice(0, 64);
  1011→      for (let i = 0; i < c.length; i++) h = (h * 31 + c.charCodeAt(i)) >>> 0;
  1012→      const id = `wxcloud-${ts}-${mine ? 'u' : 'a'}-${h % 9973}`;
  1013→      if (seen.has(id)) continue;
  1014→      // v10 近似重复防护：本机实时回复开着、云端（cron/网关）也回了同一条时，主人自己那条
  1015→      // 消息会同时从两条路落线程（本机一条 + 云端拉回一条）——同角色同内容 ±150s 内已有
  1016→      // 就不重复入列（TA 的回复两边内容不同，不去重，避免误删不同回复）。
  1017→      if (mine && cur.some((x) => x.role === 'me' && Math.abs((x.time ?? 0) - ts) < 150_000 && x.content === m.content)) continue;
  1018→      // v11 列内去重：同 id（云端同一条被双写时确定性 id 相同）只入一次——不回填 seen
  1019→      // 的话两条都会进 add，本地出现重复 key（QQ 侧同款问题的微信版）。
  1020→      seen.add(id);
  1021→      add.push({ id, role: mine ? 'me' : 'peer', content: m.content.slice(0, 2000), time: ts });
  1022→    }
  1023→    if (add.length) {
  1024→      add.sort((a, b) => a.time - b.time);
  1025→      saveMsgs(login.boundContactId, [...cur, ...add]);
  1026→      // 聊天页开着时实时增量合并（与工具轮系统行同款事件）
  1027→      window.dispatchEvent(new CustomEvent('chat-stream-sysline', { detail: { sessionKey: `wx:${login.boundContactId}` } }));
  1028→    }
  1029→    return { ok: true, msg: add.length ? `已同步 ${add.length} 条到微信聊天界面` : '云端暂无新记录', added: add.length };
  1030→  } catch (e) {
  1031→    return { ok: false, msg: e instanceof Error ? e.message : String(e) };
  1032→  }
  1033→}
