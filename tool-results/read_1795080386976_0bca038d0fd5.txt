   101→// ---------------- 扫码识别（jsQR：二维码图片 → AppID/AppSecret） ----------------
   102→
   103→/**
   104→ * 解析二维码文本 → 凭证。三种支持格式：
   105→ * ① AppID:AppSecret（自定义分隔符「:」「|」「空格」）
   106→ * ② JSON（{"appId":…,"appSecret":…} / appid / clientSecret 等常见键名）
   107→ * ③ 网址查询串（?appId=…&appSecret=… 任意顺序）
   108→ */
   109→export function parseQqQrText(raw: string): { appId: string; appSecret: string } | null {
   110→  const t = raw.trim();
   111→  if (!t) return null;
   112→  // 形态一：AppID:AppSecret（自定义分隔符「:」「|」「空格」）
   113→  const m = t.match(/^([a-zA-Z0-9]{4,64})[:|\s]+([a-zA-Z0-9!@#$%^&*_\-+=]{8,200})$/);
   114→  if (m) return { appId: m[1], appSecret: m[2] };
   115→  // 形态二：JSON
   116→  if (t.startsWith('{')) {
   117→    try {
   118→      const j = JSON.parse(t) as Record<string, unknown>;
   119→      const appId = typeof j.appId === 'string' ? j.appId : typeof j.appid === 'string' ? j.appid : '';
   120→      const appSecret =
   121→        typeof j.appSecret === 'string'
   122→          ? j.appSecret
   123→          : typeof j.appsecret === 'string'
   124→          ? j.appsecret
   125→          : typeof j.clientSecret === 'string'
   126→          ? j.clientSecret
   127→          : '';
   128→      if (appId && appSecret) return { appId, appSecret };
   129→    } catch {
   130→      /* 继续 */
   131→    }
   132→  }
   133→  // 形态三：URL 查询串
   134→  const q = t.match(/[?&](appId|appid)=([^&]+)(?:&|.*?&)(appSecret|appsecret|clientSecret|client_secret|secret)=([^&]+)/);
   135→  if (q) return { appId: q[2], appSecret: q[4] };
   136→  const q2 = t.match(/[?&](appSecret|appsecret|clientSecret|client_secret|secret)=([^&]+)(?:&|.*?&)(appId|appid)=([^&]+)/);
   137→  if (q2) return { appId: q2[4], appSecret: q2[2] };
   138→  return null;
   139→}

   141→/** 解析二维码图片 → 凭证（jsQR 解码后走 parseQqQrText）：拍照或选图均可 */
   142→export async function parseQqQrImage(file: File): Promise<{ appId: string; appSecret: string } | null> {
   143→  const { default: jsQR } = await import('jsqr');
   144→  const bitmap = await createImageBitmap(file).catch(() => null);
   145→  if (!bitmap) return null;
   146→  const canvas = document.createElement('canvas');
   147→  canvas.width = bitmap.width;
   148→  canvas.height = bitmap.height;
   149→  const ctx = canvas.getContext('2d', { willReadFrequently: true });
   150→  if (!ctx) return null;
   151→  ctx.drawImage(bitmap, 0, 0);
   152→  const data = ctx.getImageData(0, 0, canvas.width, canvas.height);
   153→  const res = jsQR(data.data, canvas.width, canvas.height);
   154→  bitmap.close?.();
   155→  if (!res || !res.data) return null;
   156→  return parseQqQrText(res.data);
   157→}

   214→function b64ToBytes(s: string): Uint8Array {
   215→  const bin = atob(s);
   216→  const out = new Uint8Array(bin.length);
   217→  for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
   218→  return out;
   219→}

   221→function bytesToB64(b: Uint8Array): string {
   222→  let s = '';
   223→  for (let i = 0; i < b.length; i++) s += String.fromCharCode(b[i]);
   224→  return btoa(s);
   225→}

   227→/** 解密 bot_encrypt_secret（AES-256-GCM：nonce=前12B、tag=后16B）→ AppSecret */
   228→async function decryptBotSecret(keyB64: string, encB64: string): Promise<string | null> {
   229→  try {
   230→    const rawKey = b64ToBytes(keyB64);
   231→    const enc = b64ToBytes(encB64);
   232→    if (rawKey.length !== 32 || enc.length < 29) return null;
   233→    const nonce = enc.subarray(0, 12);
   234→    const ctWithTag = enc.subarray(12);
   235→    const key = await crypto.subtle.importKey('raw', rawKey, { name: 'AES-GCM' }, false, ['decrypt']);
   236→    const out = await crypto.subtle.decrypt({ name: 'AES-GCM', iv: nonce }, key, ctWithTag);
   237→    return new TextDecoder().decode(out);
   238→  } catch {
   239→    return null;
   240→  }
   241→}

   281→async function qqTokenViaProxy(appId: string, appSecret: string): Promise<string> {
   282→  const res = await fetch('/api/qq-bot', {
   283→    method: 'POST',
   284→    headers: { 'Content-Type': 'application/json' },
   285→    body: JSON.stringify({ action: 'token', appId, appSecret }),

   291→    j = JSON.parse(text);
   292→  } catch {
   293→    throw new Error(`QQ 平台响应异常（${res.status}）`);
   294→  }
   295→  const rec = j as { access_token?: string; message?: string; error?: string; code?: number };
   296→  if (!res.ok || !rec.access_token) {
   297→    // 官方错误码表（业务错误 HTTP 仍 200 + code）：转成可读提示
   298→    if (rec.code === 100016) throw new Error('AppID 或 AppSecret 不正确（QQ 平台错误码 100016）：请到 q.qq.com 控制台核对凭证');
   299→    if (rec.code === 100007) throw new Error('AppID 无效或机器人状态异常（QQ 平台错误码 100007）');
   300→    if (rec.code === 10004) throw new Error('AppID 对应的机器人不存在（QQ 平台错误码 10004）');
   301→    if (rec.code === 100001) throw new Error('QQ 平台频控（100001）：请求过于频繁，稍后再试');
   302→    if (res.status === 503) throw new Error('QQ 平台拒绝了服务器的出口 IP（503）：请稍后重试');
   303→    throw new Error(rec.message || rec.error || `QQ 平台返回 ${res.status}`);
   304→  }
   305→  return rec.access_token;
   306→}

   377→      }
   378→    }
   379→    const list = Array.isArray(j.msgs) ? j.msgs : [];
   380→    const cur = loadMsgs(login.boundContactId);
   381→    // v11 合并去重：① 对本地已有 id 去重；② 云端列表内部自身去重（旧版云端同一条被
   382→    // 双写过、qq-history.json 里同 id 两份——都进 add 的话本地会出现两个同 key，QQ 聊天
   383→    // 界面同一条消息渲染两遍 + React「duplicate key」报错的直接根因）；③ 主人的消息
   384→    // 近重复防护（同微信 v10 口径）：本站 UI 发过 + 真机 QQ 又发的同内容 ±150s 只入一次。
   385→    const seen = new Set(cur.map((m) => m.id));
   386→    const add: { id: string; role: 'me' | 'peer'; content: string; time: number }[] = [];
   387→    for (const m of list) {
   388→      if (!m || typeof m.id !== 'string' || !m.id) continue;
   389→      const role: 'me' | 'peer' = m.role === 'me' || m.role === 'user' ? 'me' : 'peer';
   390→      const time = Number(m.ts) || Date.now();
   391→      const content = String(m.content ?? '').slice(0, 2000);
   392→      if (seen.has(m.id)) continue;
   393→      if (
   394→        role === 'me' &&
   395→        cur.some((x) => x.role === 'me' && Math.abs((x.time ?? 0) - time) < 150_000 && x.content === content)
   396→      ) {
   397→        continue;
   398→      }
   399→      seen.add(m.id);
   400→      add.push({ id: m.id, role, content, time });
   401→    }
   402→    if (add.length) {
   403→      add.sort((a, b) => a.time - b.time);
   404→      saveMsgs(login.boundContactId, [...cur, ...add]);
   405→      window.dispatchEvent(new CustomEvent('chat-stream-sysline', { detail: { sessionKey: `qq:${login.boundContactId}` } }));
   406→    }
   407→    return { ok: true, msg: add.length ? `已同步 ${add.length} 条 QQ 聊天记录` : '云端暂无新记录', added: add.length };
   408→  } catch (e) {
   409→    return { ok: false, msg: e instanceof Error ? e.message : String(e) };
   410→  }
   411→}

   484→  /** 云端验证结果落盘（localStorage 轻持久化——状态说明无秘密，不值得上密文 kv） */
   485→  persistCloudVerify(v: { ok: boolean; msg: string; at: number } | null): void {
   486→    try {
   487→      if (v) localStorage.setItem('qq-cloud-verify', JSON.stringify(v));
   488→      else localStorage.removeItem('qq-cloud-verify');
   489→    } catch { /* 隐私模式等忽略 */ }
   490→  },
   491→
   492→  test: async (appId, appSecret) => {
   493→    const id = appId.trim();
   494→    const secret = appSecret.trim();
   495→    if (!/^[a-zA-Z0-9]{4,64}$/.test(id)) {
   496→      const r = { ok: false, msg: 'AppID 格式不对（应为数字/字母组合，在 q.qq.com 控制台获取）' };
   497→      set({ lastTest: r });
   498→      return r;
   499→    }
   500→    if (!secret || secret.length < 8) {
   501→      const r = { ok: false, msg: 'AppSecret 不能为空（在 q.qq.com 控制台获取）' };
   502→      set({ lastTest: r });
   503→      return r;
   504→    }
   505→    set({ testing: true });
   506→    try {
   507→      let token = await qqTokenDirect(id, secret);
   508→      if (!token) {
   509→        try {
   510→          token = await qqTokenViaProxy(id, secret);
   511→        } catch (e) {
   512→          const reason = e instanceof Error ? e.message : String(e);
   513→          // 新端点（api.bot.qq.com）海外可达：两路都失败多半是业务错（凭证不对）或纯网络问题
   514→          const isBiz = /100016|100007|10004|不正确|不存在|无效/.test(reason);
   515→          const r: QqBotTestResult = isBiz
   516→            ? { ok: false, msg: `连不上 QQ 平台（${reason.slice(0, 100)}）：请到 q.qq.com 控制台核对 AppID/AppSecret 后重试。` }
   517→            : {
   518→                ok: false,
   519→                msg: `两路都连不上 QQ 机器人平台（${reason.slice(0, 80)}）。凭证已保存；部署云端助手后，到本页点「云端验证」也可确认凭证真伪。`,
   520→              };
   521→          // 网络不通 ≠ 凭证错误：保存凭证但不标记成功
   522→          const saved: QqBotLogin = { ...(memQqBot ?? { connectedAt: Date.now() }), appId: id, appSecret: secret, connectedAt: Date.now() };
   523→          memQqBot = saved;
   524→          await persistLogin(saved);
   525→          set({ testing: false, lastTest: r, login: saved });
   526→          cloudAutoSync('QQ 凭证已保存');
   527→          return r;
   528→        }
   529→      }
   530→      const me = token ? await qqBotMe(token) : null;
   531→      const saved: QqBotLogin = {
   532→        ...(memQqBot ?? {}),
   533→        appId: id,
   534→        appSecret: secret,
   535→        botName: me?.username || me?.nickname || memQqBot?.botName || undefined,
   536→        botId: me?.id || memQqBot?.botId || undefined,
   537→        connectedAt: Date.now(),
   538→      };
   539→      memQqBot = saved;
   540→      await persistLogin(saved);
   541→      const r: QqBotTestResult = {
   542→        ok: true,
   543→        msg: saved.botName ? `连接成功：机器人「${saved.botName}」凭证有效` : '连接成功，凭证有效',
   544→      };
   545→      set({ testing: false, lastTest: r, login: saved });
   546→      cloudAutoSync('QQ 接入成功');
   547→      return r;
   548→    } catch (e) {
   549→      const r: QqBotTestResult = { ok: false, msg: e instanceof Error ? e.message : String(e) };
   550→      set({ testing: false, lastTest: r });
   551→      return r;
   552→    }
   553→  },
   554→
   555→  /**
   556→   * 云端验证凭证：bots.qq.com 的 token 验证口拦海外 IP（浏览器直连与 /api 代理都拿不到
   557→   * 真伪），交给云端助手 verify-qq 动作代验（api.bot.qq.com 新端点对海外出口可达）。
   558→   * 回包 {ok, error, botName}：成功合并 botName 进凭证持久化（同 test 动作），失败用云端
   559→   * 给出的 error（云端能拿到 QQ 平台的真实报错）截 120 字展示。
   560→   */
   561→  verifyViaCloud: async () => {
   562→    const login = memQqBot;
   563→    if (!login) {
   564→      const r: QqBotTestResult = { ok: false, msg: '还没接入 QQ 机器人' };
   565→      set({ lastTest: r });
   566→      return r;
   567→    }
   568→    const cloud = wxCloudSnapshot();
   569→    const supa = cloud.supabase;
   570→    if (!(supa?.ref && supa.deployedAt)) {
   571→      const r: QqBotTestResult = { ok: false, msg: '云端还没部署：先到 设置 → 云端助手 完成部署' };
   572→      set({ lastTest: r });
   573→      return r;
   574→    }
   575→    if (get().verifying) return { ok: false, msg: '正在验证中，请稍候' };
   576→    set({ verifying: true });
   577→    try {
   578→      // Supabase 函数（v3：func-call 同源代理——存量旧版无 CORS 头的函数也能调）
   579→      const res = await fetch('/api/cloud-deploy/supabase', {
   580→        method: 'POST',
   581→        headers: { 'Content-Type': 'application/json' },
   582→        body: JSON.stringify({
   583→          op: 'func-call',
   584→          ref: supa!.ref,
   585→          secret: supa!.secret,
   586→          action: 'verify-qq',
   587→          body: { appId: login.appId, appSecret: login.appSecret },
   588→        }),
   589→        signal: AbortSignal.timeout(80_000),
   590→      });
   591→      const text = await res.text();
   592→      let j: { ok?: boolean; error?: string; message?: string; botName?: string } | null = null;
   593→      try {
   594→        j = JSON.parse(text);
   595→      } catch {
   596→        /* 非 JSON */
   597→      }
   598→      if (res.ok && j?.ok) {
   599→        const next: QqBotLogin = { ...login, botName: j.botName || login.botName || undefined };
   600→        memQqBot = next;
   601→        await persistLogin(next);
   602→        const r: QqBotTestResult = {
   603→          ok: true,
   604→          msg: `云端验证通过${j.botName ? `：机器人「${j.botName}」` : ''}，凭证有效`,
   605→        };
   606→        set({ lastTest: r, login: next, lastCloudVerify: { ok: true, msg: r.msg, at: Date.now() } });
   607→        get().persistCloudVerify({ ok: true, msg: r.msg, at: Date.now() });
   608→        cloudAutoSync('QQ 云端验证通过');
   609→        return r;
   610→      }
   611→      if (res.status === 401) {
   612→        const r: QqBotTestResult = { ok: false, msg: '任务密钥不匹配：到 设置 → 云端助手 重新点「一键部署」（或 QQ 境内出口重部署）后再试' };
   613→        set({ lastTest: r, lastCloudVerify: { ok: false, msg: r.msg, at: Date.now() } });
   614→        get().persistCloudVerify({ ok: false, msg: r.msg, at: Date.now() });
   615→        return r;
   616→      }
   617→      if (res.status === 504) {
   618→        const r: QqBotTestResult = { ok: false, msg: '连不上云端助手：请确认已一键部署（刚部署需 10-30 秒装载，稍等再试）' };
   619→        set({ lastTest: r, lastCloudVerify: { ok: false, msg: r.msg, at: Date.now() } });
   620→        get().persistCloudVerify({ ok: false, msg: r.msg, at: Date.now() });
   621→        return r;
   622→      }
   623→      const fallback = text ? `云端返回 ${res.status}：${text.slice(0, 60)}` : `云端返回 ${res.status}（云端助手可能还没更新到支持 verify-qq 的版本）`;
   624→      const r: QqBotTestResult = { ok: false, msg: (j?.error || j?.message || fallback).slice(0, 120) };
   625→      set({ lastTest: r, lastCloudVerify: { ok: false, msg: r.msg, at: Date.now() } });
   626→      get().persistCloudVerify({ ok: false, msg: r.msg, at: Date.now() });
   627→      return r;
   628→    } catch (e) {
   629→      const timedOut = e instanceof DOMException && e.name === 'TimeoutError';
   630→      const detail = e instanceof Error ? e.message : String(e);
   631→      const r: QqBotTestResult = timedOut
   632→        ? { ok: false, msg: '云端验证超时（80 秒）：请重试；多次超时请到 设置 → 云端助手 重新部署最新版' }
   633→        : { ok: false, msg: `连不上云端助手（${detail.slice(0, 80)}）：请稍后重试` };
   634→      set({ lastTest: r, lastCloudVerify: { ok: false, msg: r.msg, at: Date.now() } });
   635→      get().persistCloudVerify({ ok: false, msg: r.msg, at: Date.now() });
   636→      return r;
   637→    } finally {
   638→      set({ verifying: false });
   639→    }
   640→  },
   641→
   642→  bindContact: async (contactId, contactName) => {
   643→    const login = get().login;
   644→    if (!login) return;
   645→    const next: QqBotLogin = { ...login, boundContactId: contactId, boundContactName: contactName };
   646→    memQqBot = next;
   647→    await persistLogin(next);
   648→    set({ login: next });
   649→    // 换绑后云端人设同步更新
   650→    cloudAutoSync('QQ 换绑角色');
   651→  },
   652→
   653→  disconnect: async () => {
   654→    memQqBot = null;
   655→    await persistLogin(null);
   656→    set({ login: null, lastTest: null });
   657→    // 断开后自动重推（云端立即停用 QQ 通道）
   658→    cloudAutoSync('QQ 已断开');
   659→  },
   660→
   661→  pullHistory: async () => {
   662→    if (get().pulling) return;
   663→    set({ pulling: true });
   664→    const r = await pullQqCloudHistory();
   665→    set({ pulling: false, lastTest: r.ok ? { ok: true, msg: r.msg } : get().lastTest });
   666→  },
   667→
   668→  /**
   669→   * 官方扫码绑定：创建任务（二维码 URL）→ 2.5s 轮询 → 用户手机 QQ 确认后解密凭证 →
   670→   * 自动 test() 接入（503 地域拦时 test 内部已兜底保存凭证，提示走云端验证）。
   671→   * 二维码过期自动重建任务刷新；start/stop 用会话令牌互斥，旧循环自动退出。
   672→   */
   673→  startQrBind: async () => {
   674→    const my = ++qrSession;
   675→    set({ qr: { phase: 'starting' } });
   676→    while (qrSession === my) {
   677→      // ① 生成 AES-256 key 并创建绑定任务
   678→      const keyBuf = new Uint8Array(32);
   679→      crypto.getRandomValues(keyBuf);
   680→      const keyB64 = bytesToB64(keyBuf);
   681→      const create = await qqLiteCall('/lite/create_bind_task', { key: keyB64 });
   682→      const taskId = create?.data?.task_id;
   683→      if (qrSession !== my) return;
   684→      if (!taskId || create?.retcode !== 0) {
   685→        set({ qr: { phase: 'failed', reason: create?.msg || '连不上 q.qq.com（创建绑定任务失败）：请稍后重试' } });
   686→        return;
   687→      }
   688→      // ② 显示二维码，等待手机 QQ 扫码确认
   689→      set({ qr: { phase: 'waiting', qrUrl: qqBindConnectUrl(taskId) } });
   690→      let expired = false;
   691→      while (qrSession === my) {
   692→        await sleep(2500);
   693→        if (qrSession !== my) return;
   694→        const r = await qqLiteCall('/lite/poll_bind_result', { task_id: taskId });
   695→        if (!r) continue; // 单次网络失败继续轮询
   696→        const st = r.data?.status ?? 0;
   697→        if (st === 3) {
   698→          expired = true;
   699→          break; // 过期 → 外层重建任务自动刷新二维码
   700→        }
   701→        if (st === 2) {
   702→          // ③ 确认绑定：解密凭证并接入
   703→          set({ qr: { phase: 'saving' } });
   704→          const appId = String(r.data?.bot_appid ?? '');
   705→          const enc = r.data?.bot_encrypt_secret ?? '';
   706→          const appSecret = await decryptBotSecret(keyB64, enc);
   707→          if (qrSession !== my) return;
   708→          if (!appId || !appSecret) {
   709→            set({ qr: { phase: 'failed', reason: '凭证解密失败：请点「重新生成二维码」重试' } });
   710→            return;
   711→          }
   712→          const t = await useQqConnect.getState().test(appId, appSecret);
   713→          if (qrSession !== my) return;
   714→          if (t.ok) {
   715→            set({ qr: { phase: 'idle' }, lastTest: { ok: true, msg: '扫码接入成功：机器人凭证已保存并验证' } });
   716→            cloudAutoSync('QQ 扫码接入成功');
   717→            return;
   718→          }
   719→          // test 报 503（地域拦）：凭证其实已保存（test 内部兜底），引导云端验证
   720→          set({ qr: { phase: 'idle' }, lastTest: { ok: false, msg: t.msg } });
   721→          return;
   722→        }
   723→        // st 0/1：继续等待
   724→      }
   725→      if (expired) continue; // 过期 → 自动重新生成二维码
   726→    }
   727→  },
   728→
   729→  stopQrBind: () => {
   730→    qrSession++;
   731→    set({ qr: { phase: 'idle' } });
   732→  },
   733→}));
   734→
