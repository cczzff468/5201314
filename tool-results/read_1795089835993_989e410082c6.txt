    68→/** 接入时间格式化（YYYY-MM-DD HH:mm） */
    69→function fmtConnected(t: number): string {
    70→  const d = new Date(t);
    71→  const p = (n: number) => String(n).padStart(2, '0');
    72→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`;
    73→}
