   161→async function callLlm(cfg, messages) {

   170→          'Content-Type': 'application/json',

   295→/** 微信：轮询一轮，返回本轮处理了几条消息；TOKEN_EXPIRED 抛错。
   296→ *  v5：游标恢复 → 收一批 → 逐条回（已回过的 message_id 跳过）→ 全部成功才落游标
   297→ *  （中途失败保持旧游标：下轮重收整批，已回的靠 seen 跳过、只重试没回成功的）。 */
   298→async function tick() {
   299→  var cfg = await loadConfig(false);
   300→  if (!cfg.wechat || !cfg.wechat.botToken) return 0;
   301→  await loadWxCursor();
   302→  var d = await ilinkCall('/ilink/bot/getupdates', cfg.wechat.botToken, {
   303→    get_updates_buf: UPDATES_BUF, base_info: BASE_INFO,
   304→  });
   305→  if (d && d.error_code === -14) throw new Error('TOKEN_EXPIRED');
   306→  if (d && typeof d.get_updates_buf === 'string' && d.get_updates_buf) UPDATES_BUF = d.get_updates_buf;
   307→  var n = 0;
   308→  var pending = 0;
   309→  var msgs = (d && d.msgs) || [];
   310→  for (var i = 0; i < msgs.length; i++) {
   311→    var mid = msgs[i] && msgs[i].message_id != null ? String(msgs[i].message_id) : '';
   312→    if (wxSeenHas(mid)) continue;
   313→    pending++;
   314→    try {
   315→      if (await handleWxMsg(cfg, msgs[i])) {
   316→        n++;
   317→        if (mid) await markWxSeen(mid);
   318→      }
   319→    } catch (e) {
   320→      if (String(e && e.message).indexOf('TOKEN_EXPIRED') >= 0) throw e;
   321→      /* 单条失败继续下一条；原因已记入 WX_MSG_ERR（心跳/ping 可见） */

   457→/** 云端 TTS（OpenAI 兼容 /audio/speech，只要 wav）→ silk → QQ 原生语音条。失败返回 null（降级文字） */
   458→async function synthesizeQqVoice(cfg, text) {
   459→  var tts = cfg.qq && cfg.qq.tts;
   460→  if (!tts || !tts.baseUrl || !tts.apiKey) return null;
   461→  var b = String(tts.baseUrl).replace(/\/+$/, '');
   462→  var url = /\/audio\/speech$/.test(b) ? b : b + '/audio/speech';
   463→  try {
   464→    var r = await fetch(url, {
   465→      method: 'POST',
   466→      headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + tts.apiKey },
   467→      body: JSON.stringify({ model: tts.model || 'tts-1', voice: tts.voice || 'alloy', input: text.slice(0, 200), response_format: 'wav', speed: 1 }),
   468→      signal: AbortSignal.timeout(30000),
   469→    });
   470→    if (!r.ok) return null;
   471→    var buf = new Uint8Array(await r.arrayBuffer());
   472→    if (buf.length < 44 || String.fromCharCode(buf[0], buf[1], buf[2], buf[3]) !== 'RIFF') return null;
   473→    var silk = await silkEncode(buf);
   474→    if (!silk) return null;
   475→    var bin = '';
   476→    for (var i = 0; i < silk.length; i += 32768) bin += String.fromCharCode.apply(null, silk.subarray(i, i + 32768));
   477→    var b64 = (typeof btoa === 'function') ? btoa(bin) : Buffer.from(silk).toString('base64');
   478→    var dur = Math.max(1, Math.round(text.length / 4));
   479→    return { b64: b64, dur: dur };
   480→  } catch (e) { return null; }
   481→}

   756→/** QQ Webhook 处理（Edge Function 入口调用）：校验请求 / 事件验签+代答。返回 Response */
   757→async function handleQqWebhook(cfg, req, rawBodyText) {
   758→  var seed = qqSeed(cfg.qq.appSecret);
   759→  if (!seed) return Response.json({ error: 'no secret' }, { status: 400 });
   760→  var payload = null;
   761→  try { payload = JSON.parse(rawBodyText); } catch (e) { payload = null; }
   762→  if (!payload || typeof payload !== 'object') return Response.json({ error: 'bad json' }, { status: 400 });
   763→  var d = payload.d || payload.data;

  1006→ *  事件推送带 t/op/intent_type 等信封键——我们自己的调用一定带 action 或 secret */
  1007→function looksQqEnvelope(raw) {
  1008→  var s = String(raw || '').slice(0, 600);
  1009→  if (/"plain_token"/.test(s)) return true;
  1010→  if (/"(?:t|event_type|event|op|intent_type|id)"\\s*:\\s*"/.test(s)) return true;
  1011→  if (/"d"\\s*:\\s*\\{/.test(s) && /"(?:op|t|id)"\\s*:/.test(s)) return true;
  1012→  return false;
  1013→}

  1027→    const hasQqSig = req.headers.get('x-signature-ed25519');

  1035→      return jr({ ok: false, error: 'bad secret' }, 401);

  1050→      const authed = !!(secret && (TASK_SECRET_ENV === secret || (cfgSecret && cfgSecret === secret)));
  1051→      if (!authed) return jr({ ok: false, error: 'bad secret' }, 401);

  1075→        const content = typeof (f && f.content) === 'string' ? f.content : '';
  1076→        if (!path || path.length > 128 || !content || content.length > 800000) continue;

  1096→      return jr({ ok: false, error: 'bad secret' }, 401);
  1097→    }

  1100→    // v5：setup-cron / stop-cron 已废弃（定时任务改由手机 App 设置里的「云端轮询」开关
  1101→    // 走服务端 Management API SQL 建立/清除，不再依赖函数里的 npm:postgres）。兼容老客户端的调用

  1110→      return jr(r);
