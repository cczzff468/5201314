   105→   105→/**
   106→   106→ * 解析二维码文本 → 凭证。三种支持格式：
   107→   107→ * ① AppID:AppSecret（自定义分隔符「:」「|」「空格」）
   108→   108→ * ② JSON（{"appId":…,"appSecret":…} / appid / clientSecret 等常见键名）
   109→   109→ * ③ 网址查询串（?appId=…&appSecret=… 任意顺序）
   110→   110→ */
   111→   111→export function parseQqQrText(raw: string): { appId: string; appSecret: string } | null {
   112→   112→  const t = raw.trim();
   113→   113→  if (!t) return null;
   114→   114→  // 形态一：AppID:AppSecret（自定义分隔符「:」「|」「空格」）
   115→   115→  const m = t.match(/^([a-zA-Z0-9]{4,64})[:|\s]+([a-zA-Z0-9!@#$%^&*_\-+=]{8,200})$/);
   116→   116→  if (m) return { appId: m[1], appSecret: m[2] };
   117→   117→  // 形态二：JSON
   118→   118→  if (t.startsWith('{')) {
   119→   119→    try {
   120→   120→      const j = JSON.parse(t) as Record<string, unknown>;
   121→   121→      const appId = typeof j.appId === 'string' ? j.appId : typeof j.appid === 'string' ? j.appid : '';
   122→   122→      const appSecret =
   123→   123→        typeof j.appSecret === 'string'
   124→   124→          ? j.appSecret
   125→   125→          : typeof j.appsecret === 'string'
   126→   126→          ? j.appsecret
   127→   127→          : typeof j.clientSecret === 'string'
   128→   128→          ? j.clientSecret
   129→   129→          : '';
   130→   130→      if (appId && appSecret) return { appId, appSecret };
   131→   131→    } catch {
   132→   132→      /* 继续 */
   133→   133→    }
   134→   134→  }
   135→   135→  // 形态三：URL 查询串
   136→   136→  const q = t.match(/[?&](appId|appid)=([^&]+)(?:&|.*?&)(appSecret|appsecret|clientSecret|client_secret|secret)=([^&]+)/);
   137→   137→  if (q) return { appId: q[2], appSecret: q[4] };
   138→   138→  const q2 = t.match(/[?&](appSecret|appsecret|clientSecret|client_secret|secret)=([^&]+)(?:&|.*?&)(appId|appid)=([^&]+)/);
   139→   139→  if (q2) return { appId: q2[4], appSecret: q2[2] };
   140→   140→  return null;
   141→   141→}

   229→   229→/** 解密 bot_encrypt_secret（AES-256-GCM：nonce=前12B、tag=后16B）→ AppSecret */
   230→   230→async function decryptBotSecret(keyB64: string, encB64: string): Promise<string | null> {
   231→   231→  try {
   232→   232→    const rawKey = b64ToBytes(keyB64);
   233→   233→    const enc = b64ToBytes(encB64);
   234→   234→    if (rawKey.length !== 32 || enc.length < 29) return null;
   235→   235→    const nonce = enc.subarray(0, 12);
   236→   236→    const ctWithTag = enc.subarray(12);
   237→   237→    const key = await crypto.subtle.importKey('raw', rawKey, { name: 'AES-GCM' }, false, ['decrypt']);
   238→   238→    const out = await crypto.subtle.decrypt({ name: 'AES-GCM', iv: nonce }, key, ctWithTag);
   239→   239→    return new TextDecoder().decode(out);
   240→   240→  } catch {
   241→   241→    return null;
   242→   242→  }
   243→   243→}

   344→   344→/**
   345→   345→ * 拉取 QQ 回调生效状态（v6 自动向导）：云端函数（Supabase Edge / 腾讯云函数）在收到
   346→   346→ * QQ 平台的校验请求或事件推送时，会把 {at, kind} 写进私有桶 state/qq-callback.json——
   347→   347→ * 接入页轮询本函数，用户在 q.qq.com 控制台配完回调即自动点亮（无需手点验证）。
   348→   348→ * 返回 {ok, at?, kind?}：ok=false 且无 at = 还没配置过回调；带 at = 回调已生效。
   349→   349→ */
   350→   350→export async function fetchQqCallbackStatus(): Promise<{ ok: boolean; at?: number; kind?: string; msg?: string }> {
   351→   351→  const supa = wxCloudSnapshot().supabase;
   352→   352→  if (!supa?.ref || !supa.serviceKey) return { ok: false, msg: '云端未就绪' };
   353→   353→  try {
   354→   354→    const k = supa.serviceKey;
   355→   355→    const isNewStyle = k.startsWith('sb_secret_') || k.startsWith('sb_publishable_');
   356→   356→    const res = await fetch(`https://${supa.ref}.supabase.co/storage/v1/object/wx-cloud/state/qq-callback.json`, {
   357→   357→      headers: isNewStyle ? { apikey: k } : { apikey: k, Authorization: `Bearer ${k}` },
   358→   358→      signal: AbortSignal.timeout(12_000),
   359→   359→    });
   360→   360→    if (!res.ok) return { ok: false, msg: '还没配置过回调（配好后这里会自动点亮）' };
   361→   361→    const j = (await res.json()) as { at?: number; kind?: string };
   362→   362→    if (typeof j.at === 'number' && j.at > 0) return { ok: true, at: j.at, kind: String(j.kind ?? '') };
   363→   363→    return { ok: false, msg: '回调状态还没写入（云端函数收到 QQ 平台回调后写）' };
   364→   364→  } catch (e) {
   365→   365→    return { ok: false, msg: e instanceof Error ? e.message : String(e) };
   366→   366→  }
   367→   367→}

   371→   371→ * 返回 {ok, msg, added?}。前提：设置 → 云端助手 已完成 Supabase 连接（serviceKey 可用）。

   373→   373→export async function pullQqCloudHistory(): Promise<{ ok: boolean; msg: string; added?: number }> {

   379→   379→  if (!supa?.ref || !supa.serviceKey) return { ok: false, msg: '云端未就绪：请先到 设置 → 云端助手 完成部署' };
   380→   380→  try {
   381→   381→    // 按类型区分（官方文档）：新式 sb_secret_/sb_publishable_ key 不是 JWT，只能放
   382→   382→    // apikey 头——放 Bearer 会被 Storage 当 JWT 解析报 400 Invalid Compact JWS；eyJ 双带
   383→   383→    const k = supa.serviceKey;
   384→   384→    const isNewStyle = k.startsWith('sb_secret_') || k.startsWith('sb_publishable_');
   385→   385→    const res = await fetch(`https://${supa.ref}.supabase.co/storage/v1/object/wx-cloud/qq-history.json`, {
   386→   386→      headers: isNewStyle ? { apikey: k } : { apikey: k, Authorization: `Bearer ${k}` },

   396→   396→      .map((m) => ({ id: m.id, role: m.role, content: String(m.content ?? '').slice(0, 2000), time: m.ts || Date.now() }));
   397→   397→    if (add.length) {
   398→   398→      saveMsgs(login.boundContactId, [...cur, ...add]);
   399→   399→      window.dispatchEvent(new CustomEvent('chat-stream-sysline', { detail: { sessionKey: `qq:${login.boundContactId}` } }));
   400→   400→    }
   401→   401→    return { ok: true, msg: add.length ? `已同步 ${add.length} 条 QQ 聊天记录` : '云端暂无新记录', added: add.length };
   402→   402→  } catch (e) {
   403→   403→    return { ok: false, msg: e instanceof Error ? e.message : String(e) };
   404→   404→  }

   549→   549→  /**
   550→   550→   * 云端验证凭证：bots.qq.com 的 token 验证口拦海外 IP（浏览器直连与 /api 代理都拿不到
   551→   551→   * 真伪），交给云端助手 verify-qq 动作代验。
   552→   552→   * v6：已部署「QQ 境内出口」（腾讯云函数）时优先走它——境内 IP 直连 QQ 平台，海外出口
   553→   553→   * 被 503 拒的用户验证凭这条路恢复；未部署则回退 Supabase 函数（同 v3 逻辑）。
   554→   554→   * 回包 {ok, error, botName}：成功合并 botName 进凭证持久化（同 test 动作），失败用云端
   555→   555→   * 给出的 error（云端能拿到 QQ 平台的真实报错）截 120 字展示。
   556→   556→   */
   557→   557→  verifyViaCloud: async () => {
   558→   558→    const login = memQqBot;
   559→   559→    if (!login) {
   560→   560→      const r: QqBotTestResult = { ok: false, msg: '还没接入 QQ 机器人' };
   561→   561→      set({ lastTest: r });
   562→   562→      return r;
   563→   563→    }
   564→   564→    const cloud = wxCloudSnapshot();
   565→   565→    const tencent = cloud.tencent;
   566→   566→    const supa = cloud.supabase;
   567→   567→    if (!(tencent && tencent.deployedAt > 0) && !(supa?.ref && supa.deployedAt)) {
   568→   568→      const r: QqBotTestResult = { ok: false, msg: '云端还没部署：先到 设置 → 云端助手 完成部署' };
   569→   569→      set({ lastTest: r });
   570→   570→      return r;
   571→   571→    }
   572→   572→    if (get().verifying) return { ok: false, msg: '正在验证中，请稍候' };
   573→   573→    set({ verifying: true });
   574→   574→    try {
   575→   575→      let res: Response;
   576→   576→      if (tencent && tencent.deployedAt > 0) {
   577→   577→        // 境内出口（腾讯云函数 InvokeFunction 同源代理；动态 import 避免 store 模块环）
   578→   578→        const { funcCallTencent } = await import('./wx-cloud/tencent');
   579→   579→        const r = await funcCallTencent('verify-qq', { appId: login.appId, appSecret: login.appSecret }, 80_000);
   580→   580→        res = new Response(r.text, { status: r.status });
   581→   581→      } else {
   582→   582→        // Supabase 函数（v3：func-call 同源代理——存量旧版无 CORS 头的函数也能调）
   583→   583→        res = await fetch('/api/cloud-deploy/supabase', {
   584→   584→          method: 'POST',
   585→   585→          headers: { 'Content-Type': 'application/json' },
   586→   586→          body: JSON.stringify({
   587→   587→            op: 'func-call',
   588→   588→            ref: supa!.ref,
   589→   589→            secret: supa!.secret,
   590→   590→            action: 'verify-qq',
   591→   591→            body: { appId: login.appId, appSecret: login.appSecret },
   592→   592→          }),
   593→   593→          signal: AbortSignal.timeout(80_000),
   594→   594→        });
   595→   595→      }
   596→   596→      const text = await res.text();
   597→   597→      let j: { ok?: boolean; error?: string; message?: string; botName?: string } | null = null;
   598→   598→      try {
   599→   599→        j = JSON.parse(text);
   600→   600→      } catch {
   601→   601→        /* 非 JSON */
   602→   602→      }
   603→   603→      if (res.ok && j?.ok) {
   604→   604→        const next: QqBotLogin = { ...login, botName: j.botName || login.botName || undefined };
   605→   605→        memQqBot = next;
   606→   606→        await persistLogin(next);
   607→   607→        const r: QqBotTestResult = {
   608→   608→          ok: true,
   609→   609→          msg: `云端验证通过${j.botName ? `：机器人「${j.botName}」` : ''}，凭证有效`,
   610→   610→        };
   611→   611→        set({ lastTest: r, login: next, lastCloudVerify: { ok: true, msg: r.msg, at: Date.now() } });
   612→   612→        get().persistCloudVerify({ ok: true, msg: r.msg, at: Date.now() });
   613→   613→        cloudAutoSync('QQ 云端验证通过');
   614→   614→        return r;
   615→   615→      }
   616→   616→      if (res.status === 401) {
   617→   617→        const r: QqBotTestResult = { ok: false, msg: '任务密钥不匹配：到 设置 → 云端助手 重新点「一键部署」（或 QQ 境内出口重部署）后再试' };
   618→   618→        set({ lastTest: r, lastCloudVerify: { ok: false, msg: r.msg, at: Date.now() } });
   619→   619→        get().persistCloudVerify({ ok: false, msg: r.msg, at: Date.now() });
   620→   620→        return r;
   621→   621→      }
   622→   622→      if (res.status === 504) {
   623→   623→        const r: QqBotTestResult = { ok: false, msg: '连不上云端助手：请确认已一键部署（刚部署需 10-30 秒装载，稍等再试）' };
   624→   624→        set({ lastTest: r, lastCloudVerify: { ok: false, msg: r.msg, at: Date.now() } });
   625→   625→        get().persistCloudVerify({ ok: false, msg: r.msg, at: Date.now() });
   626→   626→        return r;
   627→   627→      }
   628→   628→      const fallback = text ? `云端返回 ${res.status}：${text.slice(0, 60)}` : `云端返回 ${res.status}（云端助手可能还没更新到支持 verify-qq 的版本）`;
   629→   629→      const r: QqBotTestResult = { ok: false, msg: (j?.error || j?.message || fallback).slice(0, 120) };
   630→   630→      set({ lastTest: r, lastCloudVerify: { ok: false, msg: r.msg, at: Date.now() } });
   631→   631→      get().persistCloudVerify({ ok: false, msg: r.msg, at: Date.now() });
   632→   632→      return r;
   633→   633→    } catch (e) {
   634→   634→      const timedOut = e instanceof DOMException && e.name === 'TimeoutError';
   635→   635→      const detail = e instanceof Error ? e.message : String(e);
   636→   636→      const r: QqBotTestResult = timedOut
   637→   637→        ? { ok: false, msg: '云端验证超时（80 秒）：请重试；多次超时请到 设置 → 云端助手 重新部署最新版' }
   638→   638→        : { ok: false, msg: `连不上云端助手（${detail.slice(0, 80)}）：请稍后重试` };
   639→   639→      set({ lastTest: r, lastCloudVerify: { ok: false, msg: r.msg, at: Date.now() } });
   640→   640→      get().persistCloudVerify({ ok: false, msg: r.msg, at: Date.now() });
   641→   641→      return r;
   642→   642→    } finally {
   643→   643→      set({ verifying: false });
   644→   644→    }
   645→   645→  },
   646→   646→
   647→   647→  bindContact: async (contactId, contactName) => {
   648→   648→    const login = get().login;
   649→   649→    if (!login) return;
   650→   650→    const next: QqBotLogin = { ...login, boundContactId: contactId, boundContactName: contactName };
   651→   651→    memQqBot = next;
   652→   652→    await persistLogin(next);
   653→   653→    set({ login: next });
   654→   654→    // 换绑后云端人设同步更新
   655→   655→    cloudAutoSync('QQ 换绑角色');
   656→   656→  },
   657→   657→
   658→   658→  disconnect: async () => {
   659→   659→    memQqBot = null;

   690→   690→        set({ qr: { phase: 'failed', reason: create?.msg || '连不上 q.qq.com（创建绑定任务失败）：请稍后重试' } });
