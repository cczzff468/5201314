   123→async function ilink<T = Record<string, unknown>>(
   124→  path: string,
   125→  opts: { botToken?: string; body?: unknown; method?: 'GET' | 'POST'; host?: string; timeoutMs?: number } = {},
   126→): Promise<T> {
   127→  const res = await fetch('/api/weixin-ilink', {
   128→    method: 'POST',
   129→    headers: { 'Content-Type': 'application/json' },
   130→    body: JSON.stringify({
   131→      path,
   132→      method: opts.method ?? 'POST',
   133→      botToken: opts.botToken,
   134→      body: opts.body ?? {},
   135→      ...(opts.host ? { host: opts.host } : {}),
   136→    }),
   137→    // 状态长轮询与 getupdates 服务端挂 ~35s：客户端窗留足 50s，其余 40s
   138→    signal: AbortSignal.timeout(opts.timeoutMs ?? 40_000),
   139→  });
   140→  const text = await res.text();
   141→  let j: unknown;
   142→  try {
   143→    j = JSON.parse(text);
   144→  } catch {
   145→    throw new Error(`iLink 响应异常（${res.status}）：${text.slice(0, 120)}`);
   146→  }
   147→  if (!res.ok) {
   148→    const rec = j as { error?: string; message?: string };
   149→    throw new Error(rec.message || rec.error || `iLink 请求失败（${res.status}）`);
   150→  }
   151→  return j as T;
   152→}

   254→/** 微信经典表情代码名全集（wxb- 内置清单的 meaning；真机微信客户端会把 [名字] 文本渲染成真表情） */
   255→const WX_CLASSIC_EMOJI_SET = new Set(BUILTIN_WX_STICKERS.map((s) => s.meaning));
   256→
   257→/** 纯表情净化后的兜底颜文字（回复整条只剩大表情标记时，不至于变空消息） */
   258→const WX_EMOJI_FALLBACKS = ['(¬‿¬)', '(๑•̀ㅂ•́)و✧', '٩(˃ᴗ˂)۶', '(っ˘ω˘ς)'];
   259→
   260→/**
   261→ * 真机微信出站净化：大表情标记 → 经典表情代码（能发的最大化），发不了的剥除。
   262→ * - [表情包:wxb-调皮] / [表情包:wwb-调皮] / [表情:调皮] 等标记：去 ID 前缀后名字在经典
   263→ *   表情清单内 → 转 [调皮]（真机微信直接显示成真表情，AI 等于真会发表情了）；不在清单
   264→ *   （用户收藏大表情）→ 剥除（文字通道发不了图，留着只会是奇怪文字）
   265→ * - [图片:xx] / [照片:xx] / [语音:xx] 等媒体标记 → 剥除
   266→ * - 已是 [微笑] 形态的经典代码零改动；净空后拿颜文字兑底
   267→ * 根因（用户实拍）：本机历史里站内微信的 [表情包:ID] 消息教会了 LLM 这个格式，而真机
   268→ * 通道发不出去——提示词拦不住，必须在发送边界确定性转换。
   269→ */
   270→export function sanitizeWxOutText(text: string): string {
   271→  const tagRe = /[\[【]\s*(?:发送了|发出了)?\s*(?:表情包|表情|图片|照片|语音|贴纸|Sticker)\s*[:：]\s*([^\]】]{1,24})\s*[\]】]/g;
   272→  let out = text.replace(tagRe, (_all, rawName: string) => {
   273→    const name = String(rawName)
   274→      .trim()
   275→      .replace(/^(?:wwb|wxb|qqb|wx|qq|mt|wb)-/i, '')
   276→      .trim();
   277→    return WX_CLASSIC_EMOJI_SET.has(name) ? `[${name}]` : '';
   278→  });
   279→  out = out
   280→    .replace(/\n{3,}/g, '\n\n')
   281→    .replace(/^[\s\n]+|[\s\n]+$/g, '')
   282→    .replace(/^[，,。.\s]+/, '');
   283→  if (out) return out;
   284→  return WX_EMOJI_FALLBACKS[Math.floor(Math.random() * WX_EMOJI_FALLBACKS.length)];
   285→}

   358→  if (!text || !msg.from_user_id || !msg.context_token) return;

   365→    ts: Date.now(),
   366→  };
   367→  st.pushMsg(entry);
   368→  // 对方（通常是你自己）说的 → 线程里记为「我方(me)」消息
   369→  appendThreadMsg(login.boundContactId, { id: `wxr-${entry.id}`, role: 'me', content: text.slice(0, 2000), time: Date.now() });
   370→  lastCtx = { fromUserId: msg.from_user_id, contextToken: msg.context_token };
   371→
   372→  // 「正在输入」：getconfig 拿 ticket，回复期间每 4.5s 续一次（host 跟随 login.baseUrl）
   373→  const api = { host: hostOf(login.baseUrl) };
   374→  let typingTicket = '';
   375→  let typingTimer: ReturnType<typeof setInterval> | null = null;
   376→  try {
   377→    const cfg = await ilink<{ ret?: number; typing_ticket?: string }>('/ilink/bot/getconfig', {
   378→      botToken: login.botToken,
   379→      host: api.host,
   380→      body: { ilink_user_id: msg.from_user_id, context_token: msg.context_token, base_info: BASE_INFO },
   381→    });
   382→    typingTicket = cfg.typing_ticket ?? '';
   383→  } catch {
   384→    /* 打字状态可有可无 */
   385→  }
   386→  const stopTyping = () => {
   387→    if (typingTimer) clearInterval(typingTimer);
   388→    typingTimer = null;
   389→    if (typingTicket) {
   390→      void ilink('/ilink/bot/sendtyping', {
   391→        botToken: login.botToken,
   392→        host: api.host,
   393→        body: { ilink_user_id: msg.from_user_id, typing_ticket: typingTicket, status: 2, base_info: BASE_INFO },
   394→      }).catch(() => undefined);
   395→    }
   396→  };
   397→  if (typingTicket) {
   398→    const beat = () =>
   399→      void ilink('/ilink/bot/sendtyping', {
   400→        botToken: login.botToken,
   401→        host: api.host,
   402→        body: { ilink_user_id: msg.from_user_id, typing_ticket: typingTicket, status: 1, base_info: BASE_INFO },
   403→      }).catch(() => undefined);
   404→    beat();
   405→    typingTimer = setInterval(beat, 4500);
   406→  }
   407→
   408→  try {
   409→    // 组 prompt：人设 system + 绑定联系人线程的最近历史（真机/网页两侧聊天都在线程里，天然互通）
   410→    const thread = loadMsgs(login.boundContactId).filter((m) => !m.kind || m.kind === 'text').slice(-14);
   411→    const history = thread.slice(0, -1).map((m) => ({ role: m.role === 'me' ? ('user' as const) : ('assistant' as const), content: m.content.slice(0, 800) }));
   412→    const contextText = thread.map((m) => `${m.role === 'me' ? '我' : 'TA'}: ${m.content}`).join('\n').slice(-1500);
   413→    const system = await buildSystemPrompt(login, contextText);
   414→    const reply = sanitizeWxOutText((await callLlm([...history, { role: 'user', content: text.slice(0, 2000) }])).trim());
   415→    if (!reply) throw new Error('生成了空回复');
   416→
   417→    await ilink('/ilink/bot/sendmessage', {
   418→      botToken: login.botToken,
   419→      host: api.host,
   420→      body: {
   421→        msg: {
   422→          from_user_id: '',
   423→          to_user_id: msg.from_user_id,
   424→          client_id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   425→          message_type: 2,
   426→          message_state: 2,
   427→          context_token: msg.context_token,
   428→          item_list: [{ type: 1, text_item: { text: reply.slice(0, 2000) } }],
   429→        },
   430→        base_info: BASE_INFO,
   431→      },
   432→    });
   433→    st.patchMsg(entry.id, { reply: reply.slice(0, 2000) });
   434→    // 角色说的 → 线程里记为「对方(peer)」，并标记为我们转发的（别再往微信发一遍）
   435→    const peerId = `wxr-${entry.id}-r`;
   436→    knownPeerIds.add(peerId);
   437→    appendThreadMsg(login.boundContactId, { id: peerId, role: 'peer', content: reply.slice(0, 2000), time: Date.now() });
   438→  } catch (e) {
   439→    st.patchMsg(entry.id, { replyErr: e instanceof Error ? e.message : String(e) });
   440→  } finally {
   441→    stopTyping();
   442→  }
   443→}

   445→// ---------------- 网页侧新回复 → 转发回真机微信 ----------------

   447→/** 已知「对方(peer)」消息 id 集合：我们自己落的不转发；boot 时把存量全部视为已知（不发历史） */
   448→const knownPeerIds = new Set<string>();

   450→async function forwardThreadWatcher(login: WxIlinkLogin): Promise<void> {
   451→  try {
   452→    const list = loadMsgs(login.boundContactId);
   453→    const fresh = list.filter((m) => m.role === 'peer' && (!m.kind || m.kind === 'text') && !knownPeerIds.has(m.id));
   454→    for (const m of fresh) {
   455→      knownPeerIds.add(m.id);
   456→      if (!lastCtx) continue; // 还没有任何微信来消息，没有 context_token 可用
   457→      await ilink('/ilink/bot/sendmessage', {
   458→        botToken: login.botToken,
   459→        host: hostOf(login.baseUrl),
   460→        body: {
   461→          msg: {
   462→            from_user_id: '',
   463→            to_user_id: lastCtx.fromUserId,
   464→            client_id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   465→            message_type: 2,
   466→            message_state: 2,
   467→            context_token: lastCtx.contextToken,
   468→            item_list: [{ type: 1, text_item: { text: sanitizeWxOutText(m.content).slice(0, 2000) } }],
   469→          },
   470→          base_info: BASE_INFO,
   471→        },
   472→      }).catch(() => undefined);
   473→    }
   474→  } catch {
   475→    /* 转发尽力而为 */
   476→  }
   477→}

   479→// ---------------- 本机轮询主循环 ----------------
   480→
   481→async function runLoop(gen: number): Promise<void> {
   482→  const st = useWxConnect.getState();
   483→  st.setLoop('running', '');
   484→  while (gen === loopGen) {
   485→    const { login, autoReply } = useWxConnect.getState();
   486→    if (!login || !autoReply) break;
   487→    try {
   488→      const d = await ilink<{ ret?: number; error_code?: number; msgs?: ILinkMessage[]; get_updates_buf?: string }>(
   489→        '/ilink/bot/getupdates',
   490→        // host 跟随 confirmed 回包的 baseurl（换域后继续可用）；长轮询窗 50s
   491→        { botToken: login.botToken, host: hostOf(login.baseUrl), timeoutMs: 50_000, body: { get_updates_buf: updatesBuf, base_info: BASE_INFO } },
   492→      );
   493→      if (gen !== loopGen) break;
   494→      if (d.error_code === -14) {
   495→        useWxConnect.getState().setLoop('error', 'Token 已过期，请回到微信重新扫码接入');
   496→        return;
   497→      }
   498→      if (typeof d.get_updates_buf === 'string' && d.get_updates_buf) updatesBuf = d.get_updates_buf;
   499→      for (const m of d.msgs ?? []) {
   500→        if (gen !== loopGen) break;
   501→        await handleIncoming(useWxConnect.getState(), login, m).catch(() => undefined);
   502→      }
   503→      // 网页侧新回复转发（每轮一次）
   504→      await forwardThreadWatcher(login);
   505→      useWxConnect.setState({ lastPollAt: Date.now() });
   506→      if (useWxConnect.getState().loopStatus === 'error') {
   507→        useWxConnect.getState().setLoop('running', '');
   508→      }
   509→    } catch (e) {
   510→      if (gen !== loopGen) return;
   511→      useWxConnect.getState().setLoop('error', e instanceof Error ? e.message : String(e));
   512→      await sleep(5000);
   513→    }
   514→  }
   515→  if (gen === loopGen) useWxConnect.getState().setLoop('off', '');
   516→}

   520→interface WxConnectStore {
   521→  ready: boolean;
   522→  login: WxIlinkLogin | null;
   523→  autoReply: boolean;
   524→  loopStatus: 'off' | 'running' | 'error';
   525→  loopError: string;
   526→  lastPollAt: number | null;
   527→  msgs: WxIlinkMsg[];
   528→  qr: WxIlinkQr | null;
   529→  qrLoading: boolean;
   530→  testing: boolean;
   531→  lastTest: { ok: boolean; msg: string; at: number } | null;
   532→  /** 云端聊天记录拉取中（拉取云端代答的真机微信对话合进手机聊天界面） */
   533→  pulling: boolean;
   534→
   535→  boot: () => Promise<void>;
   536→  /** 选好绑定联系人后取二维码并开始官方协议状态机（长轮询 + PIN + 重定向 + 换码） */
   537→  startQr: (contactId: string, contactName: string) => Promise<void>;
   538→  /** 提交微信 PIN（need_verifycode 时 UI 输入 → 下一轮轮询带 verify_code） */

   540→  cancelQr: () => void;
   541→  /** 换绑角色（不重新扫码，只改绑定的联系人） */
   542→  rebindContact: (contactId: string, contactName: string) => Promise<void>;
   543→  disconnect: () => Promise<void>;
   544→  /** 测试连接：getupdates 一次（有效 token 会快速返回，无效/过期会报错） */
   545→  test: () => Promise<boolean>;
   546→  setAutoReply: (v: boolean) => void;
   547→  pushMsg: (m: WxIlinkMsg) => void;
   548→  patchMsg: (id: string, patch: Partial<WxIlinkMsg>) => void;
   549→  clearMsgs: () => void;
   550→  /** 拉取云端微信聊天记录（测试云端/云端轮询代答的对话同步进手机聊天界面） */
   551→  pullHistory: () => Promise<void>;
   552→  setLoop: (s: 'off' | 'running' | 'error', err: string) => void;
   553→}
   554→
   555→/** 扫码状态机代数：start/stop/cancel 递增，旧循环自动退出 */
   556→let qrGen = 0;
   557→/** PIN 待提交的 resolver（need_verifycode 时 UI 输入后唤醒轮询循环） */
   558→let pinResolve: ((pin: string) => void) | null = null;
   559→
   560→export const useWxConnect = create<WxConnectStore>((set, get) => ({
   561→  ready: false,
   562→  login: null,
   563→  autoReply: false,
   564→  loopStatus: 'off',
   565→  loopError: '',
   566→  lastPollAt: null,
   567→  msgs: [],
   568→  qr: null,
   569→  qrLoading: false,
   570→  testing: false,
   571→  lastTest: null,
   572→  pulling: false,
   573→
   574→  boot: async () => {
   575→    if (wxConnectReady) return;
   576→    wxConnectReady = true;
   577→    try {
   578→      await ensureKvReady();
   579→      loadTokenHistory();
   580→      memLogin = await readEncJson<WxIlinkLogin>(LOGIN_KEY);
   581→      if (memLogin?.botToken) rememberToken(memLogin.botToken);
   582→      const msgs = await readEncJson<WxIlinkMsg[]>(MSGS_KEY).catch(() => null);
   583→      const auto = (() => {
   584→        try {
   585→          return kvGet<string>(AUTO_KEY) === '1';
   586→        } catch {
   587→          return false;
   588→        }
   589→      })();
   590→      const list = Array.isArray(msgs) ? msgs.slice(-MSGS_CAP) : [];
   591→      set({ login: memLogin, msgs: list, autoReply: auto && !!memLogin, ready: true });
   592→      if (memLogin) {
   593→        // 存量对方消息全部视为已知（不把历史回复重发到微信）
   594→        try {
   595→          for (const m of loadMsgs(memLogin.boundContactId)) {
   596→            if (m.role === 'peer') knownPeerIds.add(m.id);
   597→          }
   598→        } catch {
   599→          /* 忽略 */
   600→        }
   601→        if (auto) {
   602→          loopGen++;
   603→          void runLoop(loopGen);
   604→        }
   605→      }
   606→    } catch {
   607→      set({ ready: true });
   608→    }
   609→  },
   610→
   611→  /**
   612→   * 官方协议状态机（Tencent/openclaw-weixin docs/protocol_zh_CN.md）：
   613→   * ① POST get_bot_qrcode?bot_type=3，请求体 {local_token_list}（本机最近 10 个 bot token——
   614→   *   后端靠它认出环境亲缘，重绑不被当陌生环境风控）
   615→   * ② GET get_qrcode_status?qrcode=… 串行长轮询（服务端挂 ~35s；超时/网络错 = wait 续轮）。
   616→   *    状态：wait / scaned / confirmed（保存 bot_token+ilink_bot_id+baseurl+ilink_user_id）/
   617→   *    expired / 含 need_verifycode（手机显示数字 → 用户输入 → 下一轮带 verify_code，
   618→   *    再遇 need_verifycode = 输错重输）/ verify_code_blocked（错太多，换码重试 ≤3）/
   619→   *    scaned_but_redirect（后续轮询切 redirect_host）/ binded_redirect（已绑其它环境，终态引导）
   620→   * ③ 过期/锁死自动换码重试 3 次，超限提示稍后再试。
   621→   */
   622→  startQr: async (contactId, contactName) => {
   623→    if (get().qrLoading) return;
   624→    const myGen = ++qrGen;
   625→    set({ qrLoading: true, qr: null });
   626→    try {
   627→      await ensureKvReady();
   628→      // 换码重试外环（expired / verify_code_blocked ≤ 3 次）
   629→      for (let refresh = 0; refresh <= 3 && qrGen === myGen; refresh++) {
   630→        const d = await ilink<{ qrcode?: string; qrcode_img_content?: string; ret?: number }>(
   631→          '/ilink/bot/get_bot_qrcode?bot_type=3',
   632→          // local_token_list：本机 token 历史（协议：最多 10 个，后端认环境亲缘用）
   633→          { body: { local_token_list: tokenHistorySnapshot() } },
   634→        );
   635→        if (qrGen !== myGen) return;
   636→        if (!d.qrcode || !d.qrcode_img_content) throw new Error('iLink 未返回二维码');
   637→        // qrcode_img_content 是授权页 URL（不是图片）——本地生成二维码 data URL
   638→        const img = await QRCode.toDataURL(d.qrcode_img_content, { width: 320, margin: 2 });
   639→        set({ qr: { qrcode: d.qrcode, img, status: 'wait' }, qrLoading: false });
   640→
   641→        const qrcode = d.qrcode;
   642→        let statusHost: string | undefined; // scaned_but_redirect 后续切 redirect_host
   643→        let pin = ''; // 暂存 PIN：仅随「下一次」轮询上报（协议约定）
   644→        let wasPin = false; // 上一轮是否处于 PIN 输入（区分首次要求与输错重输）
   645→        interface QrStatusResp {
   646→          status?: string;
   647→          bot_token?: string;
   648→          ilink_bot_id?: string;
   649→          baseurl?: string;
   650→          ilink_user_id?: string;
   651→          redirect_host?: string;
   652→        }
   653→        // 串行长轮询内环
   654→        while (qrGen === myGen) {
   655→          const cur = get().qr;
   656→          if (!cur || cur.qrcode !== qrcode) return; // 被新码/取消取代
   657→          const qs = `qrcode=${encodeURIComponent(qrcode)}${pin ? `&verify_code=${encodeURIComponent(pin)}` : ''}`;
   658→          let s: QrStatusResp | null = null;
   659→          try {
   660→            s = await ilink<QrStatusResp>(`/ilink/bot/get_qrcode_status?${qs}`, {
   661→              method: 'GET',
   662→              body: {},
   663→              host: statusHost,
   664→              timeoutMs: 50_000,
   665→            });
   666→          } catch {
   667→            // 协议：单次失败/超时按 wait 续轮（短歇后重试）
   668→            await sleep(1000);
   669→            continue;
   670→          }
   671→          if (qrGen !== myGen) return;
   672→          const status = String(s?.status || 'wait');
   673→          // ① confirmed：保存凭证（含 baseurl / ilink_user_id）
   674→          if (status === 'confirmed' && s?.bot_token && s?.ilink_bot_id) {
   675→            const login: WxIlinkLogin = {
   676→              botToken: s.bot_token,
   677→              botId: s.ilink_bot_id,
   678→              ...(typeof s.baseurl === 'string' && s.baseurl ? { baseUrl: s.baseurl } : {}),
   679→              ...(typeof s.ilink_user_id === 'string' && s.ilink_user_id ? { ilinkUserId: s.ilink_user_id } : {}),
   680→              boundContactId: contactId,
   681→              boundContactName: contactName,
   682→              connectedAt: Date.now(),
   683→            };
   684→            memLogin = login;
   685→            await writeEncJson(LOGIN_KEY, login);
   686→            rememberToken(login.botToken);
   687→            personaCache = null;
   688→            lastCtx = null;
   689→            updatesBuf = '';
   690→            pinResolve = null;
   691→            set({ login, qr: { ...cur, status: 'confirmed', pinErr: undefined }, msgs: [], lastTest: null });
   692→            // 扫码接入成功后默认开启本机自动回复（用户可关；24h 在线建议部署云端）
   693→            get().setAutoReply(true);
   694→            // 已部署过云端则自动重推运行包（微信通道即刻生效，无需手动点同步）
   695→            cloudAutoSync('微信接入成功');
   696→            return;
   697→          }
   698→          // ② 需要输入 PIN（含 need_verifycode / confirmed_but_need_verifycode）
   699→          if (status.includes('need_verifycode')) {
   700→            set({
   701→              qr: {
   702→                ...cur,
   703→                status: 'need_verifycode',
   704→                pinErr: wasPin ? '数字不对，看看微信上显示的是多少，重新输一次' : undefined,
   705→              },
   706→            });
   707→            const typed = await new Promise<string>((resolve) => {
   708→              pinResolve = resolve;
   709→            });
   710→            if (qrGen !== myGen) return;
   711→            pin = typed.trim().replace(/\D/g, '').slice(0, 8);
   712→            wasPin = true;
   713→            continue; // 下一轮带 verify_code（返回 need_verifycode 依旧 = 输错）
   714→          }
   715→          // ③ 已扫码（带过 verify_code 时表示数字已接受 → 丢弃）
   716→          if (status === 'scaned') {
   717→            pin = '';
   718→            wasPin = false;
   719→            if (cur.status !== 'scaned') set({ qr: { ...cur, status: 'scaned', pinErr: undefined } });
   720→            continue;
   721→          }
   722→          // ④ 过期 / 错码锁死：换码重试（≤3），超限退出引导
   723→          if (status === 'expired' || status === 'verify_code_blocked') {
   724→            pin = '';
   725→            pinResolve = null;
   726→            if (refresh >= 3) {
   727→              set({
   728→                qr: {
   729→                  ...cur,
   730→                  status: 'expired',
   731→                  hint: status === 'verify_code_blocked' ? '数字错太多次被锁：稍后再试，或点重新获取' : undefined,
   732→                },
   733→              });
   734→              return;
   735→            }
   736→            break; // 外环换新码
   737→          }
   738→          // ⑤ 扫码后换域：后续轮询切 redirect_host
   739→          if (status === 'scaned_but_redirect') {
   740→            const rh = String(s?.redirect_host || '').trim();
   741→            if (rh) statusHost = rh.replace(/^https?:\/\//i, '').split('/')[0].toLowerCase();
   742→            if (cur.status !== 'scaned') set({ qr: { ...cur, status: 'scaned', pinErr: undefined } });
   743→            continue;
   744→          }
   745→          // ⑥ 已绑到其它环境（终态）：给解绑引导，停止轮询
   746→          if (status === 'binded_redirect') {
   747→            pinResolve = null;
   748→            set({
   749→              qr: {
   750→                ...cur,
   751→                status: 'binded_redirect',
   752→                hint: '这个微信号已连接到其它环境（比如之前连过别处的机器人）：先在那边断开/解绑，或换一个微信号扫码',
   753→              },
   754→            });
   755→            return;
   756→          }
   757→          // ⑦ wait：继续长轮询
   758→          if (cur.status !== 'wait') set({ qr: { ...cur, status: 'wait', pinErr: undefined } });
   759→        }
   760→      }
   761→    } catch (e) {
   762→      set({ qrLoading: false });
   763→      throw e;
   764→    }
   765→  },
   766→
   767→  submitPin: (pin) => {
   768→    const r = pinResolve;
   769→    pinResolve = null;
   770→    if (r) r(pin);
   771→  },
   772→
   773→  cancelQr: () => {
   774→    qrGen++;
   775→    const r = pinResolve;
   776→    pinResolve = null;
   777→    if (r) r('');
   778→    set({ qr: null });
   779→  },
   780→
   781→  rebindContact: async (contactId, contactName) => {
   782→    const login = get().login;
   783→    if (!login) return;
   784→    const next: WxIlinkLogin = { ...login, boundContactId: contactId, boundContactName: contactName };
   785→    memLogin = next;
   786→    personaCache = null;
   787→    await writeEncJson(LOGIN_KEY, next);
   788→    set({ login: next });
   789→    // 换绑后云端人设同步更新
   790→    cloudAutoSync('微信换绑角色');
   791→  },
   792→
   793→  disconnect: async () => {
   794→    loopGen++;
   795→    qrGen++;
   796→    const r = pinResolve;
   797→    pinResolve = null;
   798→    if (r) r('');
   799→    updatesBuf = '';
   800→    personaCache = null;
   801→    lastCtx = null;
   802→    knownPeerIds.clear();
   803→    memLogin = null;
   804→    try {
   805→      await kvDel(LOGIN_KEY);
   806→      kvDel(AUTO_KEY);

   903→// ---------------- 云端聊天记录拉取（云端代答的真机微信对话合进手机聊天界面） ----------------
   904→
   905→/** 云端 wx-cloud/history.json 的条目（角色视角：user=真机微信上发消息的人、assistant=TA 的回复） */
   906→interface WxCloudHistItem {
   907→  role?: string;
   908→  content?: string;
   909→  ts?: number;
   910→}
   911→
   912→/**
   913→ * 拉取云端微信聊天记录：云端助手代答（测试云端/定时轮询代跑）收发的对话存在 Supabase
   914→ * Storage wx-cloud/history.json（最近 30 条）——拉回来按「真机微信上的人=me、TA=peer」合进
   915→ * 手机里与绑定角色的微信聊天界面（去重：确定性 id，拉过的不会重复入列）。
   916→ * 与「本机实时回复」互不干扰：本机路径的消息已直接落聊天界面；这里只补云端路径的记录
   917→ * （用户实测反馈：点「测试云端」后 TA 在真机微信里回复了，但网站聊天界面里看不到——
   918→ * 云端代答的对话只有云端有，拉一次即合进来看）。
   919→ */
   920→export async function pullWxCloudHistory(): Promise<{ ok: boolean; msg: string; added?: number }> {
   921→  const cloud = wxCloudSnapshot();
   922→  const supa = cloud.supabase;
   923→  const login = memLogin;
   924→  if (!login) return { ok: false, msg: '还没接入真机微信' };
   925→  if (!login.boundContactId) return { ok: false, msg: '还没绑定角色：请先选择一个 AI 角色' };
   926→  if (!supa?.ref || !supa.serviceKey) return { ok: false, msg: '云端未就绪：请先到 设置 → 云端助手 完成部署' };
   927→  try {
   928→    // 新式 sb_secret_ key 只能放 apikey 头（Bearer 会被当 JWT 报 400 Invalid Compact JWS）
   929→    const k = supa.serviceKey;
   930→    const isNewStyle = k.startsWith('sb_secret_') || k.startsWith('sb_publishable_');
   931→    const res = await fetch(`https://${supa.ref}.supabase.co/storage/v1/object/wx-cloud/history.json`, {
   932→      headers: isNewStyle ? { apikey: k } : { apikey: k, Authorization: `Bearer ${k}` },
   933→      signal: AbortSignal.timeout(15_000),
   934→    });
   935→    if (!res.ok) return { ok: false, msg: '云端还没有微信聊天记录（真机微信上和 TA 聊过之后才有）' };
   936→    const j = (await res.json()) as { msgs?: WxCloudHistItem[] };
   937→    const list = Array.isArray(j.msgs) ? j.msgs : [];
   938→    const cur = loadMsgs(login.boundContactId);
   939→    const seen = new Set(cur.map((m) => m.id));
   940→    const add: WxMsg[] = [];
   941→    for (const m of list) {
   942→      if (!m || typeof m.content !== 'string' || !m.content.trim()) continue;
   943→      const mine = m.role === 'user'; // 云端视角：user=真机微信上发消息的人（=机主）→ 站内 me
   944→      const ts = Number(m.ts) || Date.now();
   945→      // 确定性 id：ts+角色+内容散列——同一云端记录重复拉取不重复入列（head 截断不撞号）
   946→      let h = 0;
   947→      const c = m.content.slice(0, 64);
   948→      for (let i = 0; i < c.length; i++) h = (h * 31 + c.charCodeAt(i)) >>> 0;
   949→      const id = `wxcloud-${ts}-${mine ? 'u' : 'a'}-${h % 9973}`;
   950→      if (seen.has(id)) continue;
   951→      add.push({ id, role: mine ? 'me' : 'peer', content: m.content.slice(0, 2000), time: ts });
   952→    }
   953→    if (add.length) {
   954→      add.sort((a, b) => a.time - b.time);
   955→      saveMsgs(login.boundContactId, [...cur, ...add]);
   956→      // 聊天页开着时实时增量合并（与工具轮系统行同款事件）
   957→      window.dispatchEvent(new CustomEvent('chat-stream-sysline', { detail: { sessionKey: `wx:${login.boundContactId}` } }));
   958→    }
   959→    return { ok: true, msg: add.length ? `已同步 ${add.length} 条到微信聊天界面` : '云端暂无新记录', added: add.length };
   960→  } catch (e) {
   961→    return { ok: false, msg: e instanceof Error ? e.message : String(e) };
   962→  }
   963→}
