    29→/** z-ai 可用音色（不认识的音色名一律回 tongtong——运行包里存的是 OpenAI 音色名时也不报错） */
    30→const ZAI_VOICES = ['tongtong', 'chuichui', 'xiaochen', 'jam', 'kazi', 'douji', 'luodo'];
    31→let zaiInstance: any | null = null;
    32→
    33→/** 本地合成语音（POST /tts，OpenAI 兼容形状 {input, voice, speed}）→ wav 二进制。
    34→ *  z-ai-web-dev-sdk 仅后端使用（本 mini-service 是后端）；SDK 缺失/合成失败返回 5xx，
    35→ *  云端核心拿到非 200 自动降级纯文字（不炸收发主流程）——QQ 语音条零配置可用。 */
    36→async function handleLocalTts(req: Request): Promise<Response> {
    37→  let body: Record<string, any> = {};
    38→  try {
    39→    body = (await req.json()) as Record<string, any>;
    40→  } catch {
    41→    return Response.json({ error: 'bad json' }, { status: 400 });
    42→  }
    43→  const text = String(body.input || body.text || '').trim().slice(0, 1000);
    44→  if (!text) return Response.json({ error: 'empty text' }, { status: 400 });
    45→  const voiceRaw = String(body.voice || '').trim();
    46→  const voice = ZAI_VOICES.includes(voiceRaw) ? voiceRaw : 'tongtong';
    47→  const speedRaw = Number(body.speed);
    48→  const speed = Number.isFinite(speedRaw) ? Math.min(2, Math.max(0.5, speedRaw)) : 1;
    49→  try {
    50→    if (!zaiInstance) {
    51→      const mod: any = await import('z-ai-web-dev-sdk');
    52→      const ZAI = mod.default ?? mod;
    53→      zaiInstance = await ZAI.create();
    54→    }
    55→    const res = await zaiInstance.audio.tts.create({
    56→      input: text,
    57→      voice,
    58→      speed,
    59→      response_format: 'wav',
    60→      stream: false,
    61→    });
    62→    const buf = new Uint8Array(await res.arrayBuffer());
    63→    if (!buf.length) return Response.json({ error: 'empty audio' }, { status: 502 });
    64→    return new Response(buf, {
    65→      headers: { 'Content-Type': 'audio/wav', 'Content-Length': String(buf.length) },
    66→    });
    67→  } catch (e) {
    68→    zaiInstance = null; // 失败重置单例（下汰重试）
    69→    return Response.json({ error: 'tts failed: ' + String((e as Error)?.message || e).slice(0, 80) }, { status: 503 });
    70→  }
    71→}

   215→function scheduleReconnect(gen: number): void {
   216→  if (!rt || rt.gen !== gen || wantClose) return;
   217→  rt.qq.state = 'connecting';
   218→  rt.qq.reconnects += 1;
   219→  reconnectDelay = Math.min(Math.round(reconnectDelay * 1.6), 30000);
   220→  if (reconnectTimer) clearTimeout(reconnectTimer);
   221→  reconnectTimer = setTimeout(() => {
   222→    void connectQq(gen);
   223→  }, reconnectDelay);
   224→}

   226→async function connectQq(gen: number): Promise<void> {
   227→  if (!rt || rt.gen !== gen || wantClose) return;
   228→  // v11 并发防抖：一轮连接在飞（config→token→网关地址→WS 握手）就不再开第二条
   229→  if (qqConnecting) return;
   230→  // v11 僵尸连接防抖：上一条 socket 还活着/还在握手就不再开新的——旧版会直接把 ws 变量
   231→  // 指到新 socket，旧 socket 依旧收事件但心跳发到死 socket 上，状态永远「connecting」
   232→  if (ws && (ws.readyState === 0 || ws.readyState === 1)) return;
   233→  qqConnecting = true;
   234→  try {
   235→  const q = rt.qq;
   236→  try {
   237→    // 凭证来自用户 Supabase 运行包（loadConfig 60s 缓存；换绑/改密钥 60s 内跟上）
   238→    const cfg = await rt.core.loadConfig(true);
   239→    if (!cfg.qq || !cfg.qq.appId) {
   240→      q.state = 'disabled';
   241→      q.lastError = '';
   242→      // v10：没凭证不是终态——手机 App 接入 QQ / 自动同步运行包后这里要自己亮起来。
   243→      // 30 秒后再查一次（fetch 一次 config.json，重活轻）；旧版直接 return，接入后
   244→      // 网关永远不知道，QQ 通道永远「正在连接」，用户实测踩坑。
   245→      if (reconnectTimer) clearTimeout(reconnectTimer);
   246→      reconnectTimer = setTimeout(() => {
   247→        void connectQq(gen);
   248→      }, 30000);
   249→      return;
   250→    }
   251→    q.state = 'connecting';
   252→    const token = await rt.core.qqToken(cfg);
   253→    // 取 WSS 网关地址（官方：GET /gateway，鉴权 QQBot <access_token>）；
   254→    // v10 加超时：原来无超时，QQ 网关挂起时 connectQq 卡死 → QQ 通道永远 connecting
   255→    const gwr = await fetch(QQ_GATEWAY_API, {
   256→      headers: { Authorization: 'QQBot ' + token },
   257→      signal: AbortSignal.timeout(20000),
   258→    });
   259→    if (!gwr.ok) {
   260→      q.state = 'error';
   261→      q.lastError = `QQ 网关地址获取失败 ${gwr.status}`;
   262→      scheduleReconnect(gen);
   263→      return;
   264→    }
   265→    const gwj = (await gwr.json().catch(() => null)) as { url?: string } | null;
   266→    const wssUrl = gwj?.url;
   267→    if (!wssUrl) {
   268→      q.state = 'error';
   269→      q.lastError = 'QQ 网关地址获取失败：回包无 url';
   270→      scheduleReconnect(gen);
   271→      return;
   272→    }
   273→    wantClose = false;
   274→    ws = new WebSocket(wssUrl);
   275→    ws.onmessage = (ev: MessageEvent) => {
   276→      if (!rt || rt.gen !== gen || wantClose) return;
   277→      try {
   278→        void handleWsFrame(JSON.parse(String(ev.data)) as Record<string, any>, gen, token);
   279→      } catch {
   280→        /* 非 JSON 帧忽略 */
   281→      }
   282→    };
   283→    ws.onclose = (ev: CloseEvent) => {
   284→      stopHbTimer();
   285→      console.log('[im-gateway] QQ WS close:', ev.code, ev.reason?.slice(0, 80), 'gen', gen);
   286→      if (rt && rt.gen === gen && !wantClose) scheduleReconnect(gen);
   287→    };
   288→    ws.onerror = () => {
   289→      /* onclose 兜底重连；这里只记状态 */
   290→      if (rt && rt.gen === gen) rt.qq.state = rt.qq.sessionId ? 'connecting' : 'error';
   291→    };
   292→  } catch (e) {
   293→    q.state = 'error';
   294→    q.lastError = String((e as Error)?.message || e).slice(0, 160);
   295→    scheduleReconnect(gen);
   296→  }
   297→  } finally {
   298→    qqConnecting = false;
   299→  }
   300→}

   302→async function handleWsFrame(f: Record<string, any>, gen: number, token: string): Promise<void> {
   303→  if (!rt || rt.gen !== gen) return;
   304→  const q = rt.qq;
   305→  const op = Number(f.op);
   306→  if (typeof f.s === 'number' && f.s > 0) q.lastSeq = f.s;
   307→
   308→  if (op === 10) {
   309→    // Hello：心跳周期 + 登录鉴权（有会话则 Resume 断点续传，官方补发遗漏事件）
   310→    hbInterval = Number(f.d?.heartbeat_interval) || 41250;
   311→    stopHbTimer();
   312→    q.hbOk = true;
   313→    hbTimer = setInterval(() => {
   314→      if (!rt || rt.gen !== gen || wantClose || !ws || ws.readyState !== 1) {
   315→        stopHbTimer();
   316→        return;
   317→      }
   318→      // 上一轮心跳没收到 ACK → 连接僵死，断开走重连（Resume 续传）
   319→      if (!q.hbOk) {
   320→        try {
   321→          ws.close();
   322→        } catch {
   323→          /* 忽略 */
   324→        }
   325→        return;
   326→      }
   327→      q.hbOk = false;
   328→      wsSend({ op: 1, d: q.lastSeq || null });
   329→    }, hbInterval);
   330→    if (q.sessionId && q.lastSeq) {
   331→      wsSend({ op: 6, d: { token: 'QQBot ' + token, session_id: q.sessionId, seq: q.lastSeq } });
   332→    } else {
   333→      wsSend({
   334→        op: 2,
   335→        d: {
   336→          token: 'QQBot ' + token,
   337→          intents: QQ_INTENTS,
   338→          shard: [0, 1],
   339→          properties: { $os: 'linux', $browser: 'im-gateway', $device: 'im-gateway' },
   340→        },
   341→      });
   342→    }
   343→    return;
   344→  }
   345→  if (op === 11) {
   346→    q.hbOk = true;
   347→    // 心跳 ACK 到达即代表会话已鉴权（Resume 后平台可能不发 RESUMED——心跳在线就是真在线）
   348→    if (q.state === 'connecting') q.state = 'ready';
   349→    return;
   350→  }
   351→  if (op === 7) {
   352→    // 服务端要求重连：立即断开走 Resume
   353→    console.log('[im-gateway] QQ WS op7 reconnect 指令');
   354→    try {
   355→      ws?.close();
   356→    } catch {
   357→      /* 忽略 */
   358→    }
   359→    return;
   360→  }
   361→  if (op === 9) {
   362→    // Identify/Resume 参数错（会话失效等）：重置会话重新鉴权
   363→    q.sessionId = '';
   364→    q.state = 'connecting';
   365→    q.lastError = 'QQ 平台返回会话无效（op9），重新鉴权中';
   366→    console.log('[im-gateway] QQ WS op9 invalid session:', JSON.stringify(f.d || null).slice(0, 120));
   367→    try {
   368→      ws?.close();
   369→    } catch {
   370→      /* 忽略 */
   371→    }
   372→    return;
   373→  }
   374→  if (op === 0) {
   375→    const t = String(f.t || '');
   376→    if (t === 'READY') {
   377→      q.state = 'ready';
   378→      q.sessionId = String(f.d?.session_id || '');
   379→      q.botName = String(f.d?.user?.username || f.d?.user?.nickname || '');
   380→      q.lastError = '';
   381→      reconnectDelay = 2000;
   382→      console.log('[im-gateway] QQ WS READY:', q.botName, 'session', q.sessionId.slice(0, 8));
   383→      return;
   384→    }
   385→    if (t === 'RESUMED') {
   386→      q.state = 'ready';
   387→      q.lastError = '';
   388→      reconnectDelay = 2000;
   389→      console.log('[im-gateway] QQ WS RESUMED: session', q.sessionId.slice(0, 8), 'seq', q.lastSeq);
   390→      return;
   391→    }
   392→    if (t === 'C2C_MESSAGE_CREATE' || t === 'GROUP_AT_MESSAGE_CREATE') {
   393→      q.lastEventAt = Date.now();
   394→      q.events += 1;
   395→      const d = f.d || {};
   396→      console.log('[im-gateway] QQ event:', t, 'id=', String(d.id || '').slice(0, 18));
   397→      const key = t + ':' + String(d.id || '');
   398→      const seenAt = qqSeen.get(key);
   399→      qqSeen.set(key, Date.now());
   400→      pruneSeen();
   401→      if (seenAt) return; // 平台重推的同一条消息：不重复回复
   402→      // 消息代答复用云端核心（人设/表情包/语音条/历史回写 Supabase 全在内）；cfg 走 60s 缓存
   403→      try {
   404→        const cfg = await rt.core.loadConfig(false);
   405→        const handled = await rt.core.handleQqEvent(cfg, t, d);
   406→        console.log('[im-gateway] QQ 处理', handled ? '已回复' : '跳过（无内容）');
   407→      } catch (e) {
   408→        q.lastError = String((e as Error)?.message || e).slice(0, 160);
   409→        console.log('[im-gateway] QQ 处理失败:', q.lastError);
   410→      }
   411→      return;
   412→    }
   413→    return; // 其它事件（好友添加等）忽略
   414→  }
   415→}

   418→  wantClose = true;
   419→  stopHbTimer();
   420→  qqConnecting = false;
   421→  if (qqLockTimer) {

   437→  }
   438→  ws = null;

   442→ *  30s 后自动重查凭证（网页重新接入 → 自动点亮）。微信轮询与网关本身不受影响。 */
   443→function disableQqChannel(gen: number): void {
   444→  if (!rt || rt.gen !== gen) return;
   445→  const q = rt.qq;
   446→  stopHbTimer();
   447→  if (reconnectTimer) {
   448→    clearTimeout(reconnectTimer);
   449→    reconnectTimer = null;
   450→  }
   451→  wantClose = true; // 停在飞帧/心跳；30s 后重查前复位
   452→  try {
   453→    ws?.close();
   454→  } catch {
   455→    /* 忽略 */
   456→  }
   457→  ws = null;
   458→  q.sessionId = '';
   459→  q.state = 'disabled';
   460→  q.lastError = '';
   461→  q.hbOk = false;
   462→  reconnectDelay = 2000;
   463→  console.log('[im-gateway] QQ 凭证已从运行包移除（网页断开）→ 长连接已拆除，TA 在真机 QQ 上下线');
   464→  reconnectTimer = setTimeout(() => {
   465→    if (!rt || rt.gen !== gen) return;
   466→    wantClose = false;
   467→    void connectQq(gen);
   468→  }, 30000);
   469→}

   473→async function wxLoop(gen: number): Promise<void> {
   474→  let errs = 0;
   475→  while (rt && rt.gen === gen) {
   476→    const wx = rt.wx;
   477→    try {
   478→      const n = (await rt.core.tick()) as number;
   479→      errs = 0;
   480→      wx.lastRunAt = Date.now();
   481→      wx.lastError = '';
   482→      if (n > 0) wx.replied += n;
   483→      await rt.core.writeHeartbeat({ source: 'im-gateway' });
   484→    } catch (e) {
   485→      const msg = String((e as Error)?.message || e);
   486→      wx.lastError = msg.slice(0, 160);
   487→      wx.lastRunAt = Date.now();
   488→      if (msg.indexOf('TOKEN_EXPIRED') >= 0) {
   489→        // 微信 Token 过期：等主人回手机 App 重新扫码（期间静默，5 分钟再探一次）
   490→        await sleep(300000);
   491→      } else if (msg.indexOf('未配置') >= 0 || msg.indexOf('运行包缺失') >= 0) {
   492→        await sleep(60000);
   493→      } else {
   494→        errs++;
   495→        await sleep(Math.min(5000 * errs, 30000));
   496→      }
   497→      continue;
   498→    }
   499→    // 正常周期：getupdates 服务端挂 ~35s；空闲（未接入微信）也 10s 一探，不热转
   500→    await sleep(10000);
   501→  }
   502→}

   567→/** 探测云端函数代际：ping 动作（所有版本都无害）带 ver>=8 才支持 read-file 中继 */
   568→async function probeRelayCapability(relayUrl: string, relaySecret: string): Promise<{ ok: boolean; msg: string }> {
   569→  try {
   570→    const r = await fetch(relayUrl + '?action=ping', {
   571→      method: 'POST',
   572→      headers: { 'Content-Type': 'application/json', 'X-Task-Secret': relaySecret },
   573→      body: '{}',
   574→      signal: AbortSignal.timeout(15_000),
   575→    });
   576→    const j = (await r.json().catch(() => null)) as { ok?: boolean; ver?: number } | null;
   577→    if (r.status === 401) return { ok: false, msg: '任务密钥不匹配：请回手机 App 设置 → 云端助手 重新点「一键部署」升级函数' };
   578→    if (!j || !j.ok) return { ok: false, msg: '云端函数不可达或未部署（HTTP ' + r.status + '）——请确认已完成一键部署' };
   579→    if ((Number(j.ver) || 0) < 8) return { ok: false, msg: '云端函数是旧版（不带中继读能力）——请回手机 App 设置 → 云端助手 再点一次「一键部署」' };
   580→    return { ok: true, msg: '' };
   581→  } catch (e) {
   582→    return { ok: false, msg: '连不上云端函数：' + String((e as Error)?.message || e).slice(0, 80) };
   583→  }
   584→}

   586→async function startGateway(supaUrl: string, supaKey: string, ref: string, relayUrl: string, relaySecret: string): Promise<{ ok: boolean; msg: string }> {
   587→  const url = String(supaUrl || '').replace(/\/+$/, '');
   588→  const key = String(supaKey || '');
   589→  const relay = String(relayUrl || '').replace(/\/+$/, '');
   590→  const rsecret = String(relaySecret || '');
   591→  if (!/^https:\/\/[a-z0-9-]+\.supabase\.co$/i.test(url)) return { ok: false, msg: 'supaUrl 不是有效的 Supabase 项目地址' };
   592→  if (rt && rt.ref === ref) return { ok: true, msg: '网关已在运行' };
   593→
   594→  // v8 通道决策：优先中继（任务密钥随部署注入，永远有效；用户 Storage key 可能被轮换全拒），
   595→  // 中继不可用时兑底直连密钥；两样都没有才报错。
   596→  let mode: 'relay' | 'direct' = 'relay';
   597→  if (rsecret && relay) {
   598→    const probe = await probeRelayCapability(relay, rsecret);
   599→    if (!probe.ok) {
   600→      if (key.length >= 20) {
   601→        mode = 'direct'; // 有直连密钥：降级直连，网关仍能跑
   602→        console.log('[im-gateway] 中继不可用，降级直连：', probe.msg);
   603→      } else {
   604→        return { ok: false, msg: '网关无法读云端配置：' + probe.msg };
   605→      }
   606→    }
   607→  } else if (key.length >= 20) {
   608→    mode = 'direct';
   609→  } else {
   610→    return { ok: false, msg: '缺少云端凭证（任务密钥/项目密钥都没有）：请回手机 App 重新一键部署' };
   611→  }
   612→
   613→  // 换连接：先停旧会话再立新
   614→  if (rt) await stopGateway();
   615→
   616→  // 中继模式落盘（自恢复用；直连模式不落盘——密钥不该进磁盘）
   617→  if (mode === 'relay') {
   618→    writePersistedState({ v: 1, supaUrl: url, ref: ref || url.replace(/^https:\/\//, '').split('.')[0], relayUrl: relay, relaySecret: rsecret, savedAt: Date.now() });
   619→  } else {
   620→    clearPersistedState();
   621→  }
   622→
   623→  const core = await materializeCore(url, mode === 'relay' ? '' : key, mode === 'relay' ? relay : '', mode === 'relay' ? rsecret : '');
   624→  // 启动自检：真实读一次运行包（密钥错/包缺失立刻报可读原因，不静默空转）
   625→  try {
   626→    await core.loadConfig(true);
   627→  } catch (e) {
   628→    const msg = String((e as Error)?.message || e);
   629→    clearPersistedState();
   630→    return { ok: false, msg: '读取云端运行包失败：' + msg.slice(0, 120) };
   631→  }
   632→  const gen = rt ? rt.gen + 1 : 1;
   633→  rt = {
   634→    gen,
   635→    ref: ref || url.replace(/^https:\/\//, '').split('.')[0],
   636→    startedAt: Date.now(),
   637→    mode,
   638→    core,
   639→    qq: { state: 'connecting', sessionId: '', botName: '', lastSeq: 0, lastEventAt: 0, events: 0, reconnects: 0, lastError: '', hbOk: false },
   640→    wx: { lastRunAt: 0, lastError: '', replied: 0 },
   641→  };
   642→  // qqSeen 跨连接保留无意义：清掉
   643→  qqSeen.clear();
   644→  wantClose = false;
   645→  reconnectDelay = 2000;
   646→  void wxLoop(gen);
   647→  void connectQq(gen);
   648→  // v9：QQ 连接锁续期（每 15s）——QQ 在线期间云端函数 cron-cycle 看到新鲜锁就不另开
   649→  // 第二条连接（本网关是“本站在线时”的 QQ 持有方；本网关不在线时锁自然过期，云端接手）
   650→  if (qqLockTimer) clearInterval(qqLockTimer);
   651→  qqLockTimer = setInterval(() => {
   652→    if (!rt || rt.gen !== gen) {
   653→      if (qqLockTimer) clearInterval(qqLockTimer);
   654→      return;
   655→    }
   656→    if (rt.qq.state === 'disabled') return; // 没接 QQ 不占锁（云端/网关都无人需要）
   657→    void rt.core.refreshQqWsLock('gw').catch(() => undefined);
   658→  }, 15000);
   659→  // v12 凭证看门狗（每 30s）：QQ 在线期间重读运行包——手机 App「断开连接」后凭证段
   660→  // 被移除时拆长连接（真机 QQ 上 TA 下线）；网络读失败不打扰在线连接。
   661→  if (qqCfgTimer) clearInterval(qqCfgTimer);
   662→  qqCfgTimer = setInterval(() => {
   663→    if (!rt || rt.gen !== gen) {
   664→      if (qqCfgTimer) clearInterval(qqCfgTimer);
   665→      return;
   666→    }
   667→    if (rt.qq.state === 'disabled') return; // disabled 分支自己的 30s 重查已覆盖
   668→    void rt.core
   669→      .loadConfig(true)
   670→      .then((cfg: { qq?: { appId?: string } } | null) => {
   671→        if (!rt || rt.gen !== gen) return;
   672→        if (!cfg || !cfg.qq || !cfg.qq.appId) disableQqChannel(gen);
   673→      })
   674→      .catch(() => undefined);
   675→  }, 30000);
   676→  return { ok: true, msg: '网关已启动：QQ WebSocket 长连接 + 微信自动回复' + (mode === 'relay' ? '（云端函数中继通道）' : '') };
   677→}

   679→async function stopGateway(): Promise<{ ok: boolean; msg: string }> {
   680→  if (!rt) {
   681→    clearPersistedState();
   682→    return { ok: true, msg: '网关本就未在运行' };
   683→  }
   684→  const cur = rt;
   685→  rt = null;
   686→  stopQq();
   687→  clearPersistedState(); // v8：自恢复状态一并清掉（重启后不再自动拉起）
   688→  cur.gen = -1; // 让微信循环自然退出（下一轮 while 判定失败）
   689→  return { ok: true, msg: '网关已停止（QQ 长连接已断开、微信自动回复已停）' };
   690→}
