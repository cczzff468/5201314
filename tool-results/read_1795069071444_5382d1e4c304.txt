   123→async function ilink<T = Record<string, unknown>>(
   124→  path: string,
   125→  opts: { botToken?: string; body?: unknown; method?: 'GET' | 'POST'; host?: string; timeoutMs?: number } = {},
   126→): Promise<T> {
   127→  const res = await fetch('/api/weixin-ilink', {
   128→    method: 'POST',
   129→    headers: { 'Content-Type': 'application/json' },
   130→    body: JSON.stringify({
   131→      path,
   132→      method: opts.method ?? 'POST',
   133→      botToken: opts.botToken,
   134→      body: opts.body ?? {},
   135→      ...(opts.host ? { host: opts.host } : {}),
   136→    }),
   137→    // 状态长轮询与 getupdates 服务端挂 ~35s：客户端窗留足 50s，其余 40s
   138→    signal: AbortSignal.timeout(opts.timeoutMs ?? 40_000),
   139→  });
   140→  const text = await res.text();
   141→  let j: unknown;
   142→  try {
   143→    j = JSON.parse(text);
   144→  } catch {
   145→    throw new Error(`iLink 响应异常（${res.status}）：${text.slice(0, 120)}`);
   146→  }
   147→  if (!res.ok) {
   148→    const rec = j as { error?: string; message?: string };
   149→    throw new Error(rec.message || rec.error || `iLink 请求失败（${res.status}）`);
   150→  }
   151→  return j as T;
   152→}

   260→const WX_EMOJI_FALLBACKS = ['(¬‿¬)', '(๑•̀ㅂ•́)و✧', '٩(˃ᴗ˂)۶', '(っ˘ω˘ς)'];

   262→/**
   263→ * 真机微信出站净化：大表情标记 → 经典表情代码（能发的最大化），发不了的剥除。
   264→ * - [表情包:wxb-调皮] / [表情包:wwb-调皮] / [表情:调皮] 等标记：去 ID 前缀后名字在经典
   265→ *   表情清单内 → 转 [调皮]（真机微信直接显示成真表情，AI 等于真会发表情了）；不在清单
   266→ *   （用户收藏大表情）→ 剥除（文字通道发不了图，留着只会是奇怪文字）
   267→ * - [图片:xx] / [照片:xx] / [语音:xx] 等媒体标记 → 剥除
   268→ * - 已是 [微笑] 形态的经典代码零改动；净空后拿颜文字兑底
   269→ * 根因（用户实拍）：本机历史里站内微信的 [表情包:ID] 消息教会了 LLM 这个格式，而真机
   270→ * 通道发不出去——提示词拦不住，必须在发送边界确定性转换。

   272→export function sanitizeWxOutText(text: string): string {
   273→  const tagRe = /[\[【]\s*(?:发送了|发出了)?\s*(?:表情包|表情|图片|照片|语音|贴纸|Sticker)\s*[:：]\s*([^\]】]{1,24})\s*[\]】]/g;
   274→  let out = text.replace(tagRe, (_all, rawName: string) => {
   275→    const name = String(rawName)
   276→      .trim()
   277→      .replace(/^(?:wwb|wxb|qqb|wx|qq|mt|wb)-/i, '')
   278→      .trim();
   279→    return WX_CLASSIC_EMOJI_SET.has(name) ? `[${name}]` : '';
   280→  });
   281→  out = out
   282→    .replace(/\n{3,}/g, '\n\n')
   283→    .replace(/^[\s\n]+|[\s\n]+$/g, '')
   284→    .replace(/^[，,。.\s]+/, '');
   285→  if (out) return out;
   286→  return WX_EMOJI_FALLBACKS[Math.floor(Math.random() * WX_EMOJI_FALLBACKS.length)];
   287→}

   354→/** 处理一条来消息：落线程 → 打字状态 → 人设回复 → 发回微信 + 落线程 */
   355→async function handleIncoming(st: WxConnectStore, login: WxIlinkLogin, msg: ILinkMessage): Promise<void> {
   356→  const textItem = (msg.item_list ?? []).find((i) => i.type === 1);
   357→  const text = (textItem?.text_item?.text ?? '').trim();
   358→  if (!text || !msg.from_user_id || !msg.context_token) return;
   359→  if (msg.from_user_id === login.botId) return; // 自己发出的（多端同收）
   360→
   361→  const entry: WxIlinkMsg = {
   362→    id: `${msg.message_id ?? Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   363→    fromUserId: msg.from_user_id,
   364→    text: text.slice(0, 2000),
   365→    ts: Date.now(),
   366→  };
   367→  st.pushMsg(entry);
   368→  // 对方（通常是你自己）说的 → 线程里记为「我方(me)」消息
   369→  appendThreadMsg(login.boundContactId, { id: `wxr-${entry.id}`, role: 'me', content: text.slice(0, 2000), time: Date.now() });
   370→  lastCtx = { fromUserId: msg.from_user_id, contextToken: msg.context_token };
   371→
   372→  // 「正在输入」：getconfig 拿 ticket，回复期间每 4.5s 续一次（host 跟随 login.baseUrl）
   373→  const api = { host: hostOf(login.baseUrl) };
   374→  let typingTicket = '';
   375→  let typingTimer: ReturnType<typeof setInterval> | null = null;
   376→  try {
   377→    const cfg = await ilink<{ ret?: number; typing_ticket?: string }>('/ilink/bot/getconfig', {
   378→      botToken: login.botToken,
   379→      host: api.host,
   380→      body: { ilink_user_id: msg.from_user_id, context_token: msg.context_token, base_info: BASE_INFO },
   381→    });
   382→    typingTicket = cfg.typing_ticket ?? '';
   383→  } catch {
   384→    /* 打字状态可有可无 */
   385→  }
   386→  const stopTyping = () => {
   387→    if (typingTimer) clearInterval(typingTimer);
   388→    typingTimer = null;
   389→    if (typingTicket) {
   390→      void ilink('/ilink/bot/sendtyping', {
   391→        botToken: login.botToken,
   392→        host: api.host,
   393→        body: { ilink_user_id: msg.from_user_id, typing_ticket: typingTicket, status: 2, base_info: BASE_INFO },
   394→      }).catch(() => undefined);
   395→    }
   396→  };
   397→  if (typingTicket) {
   398→    const beat = () =>
   399→      void ilink('/ilink/bot/sendtyping', {
   400→        botToken: login.botToken,
   401→        host: api.host,
   402→        body: { ilink_user_id: msg.from_user_id, typing_ticket: typingTicket, status: 1, base_info: BASE_INFO },
   403→      }).catch(() => undefined);
   404→    beat();
   405→    typingTimer = setInterval(beat, 4500);
   406→  }
   407→
   408→  try {
   409→    // 组 prompt：人设 system + 绑定联系人线程的最近历史（真机/网页两侧聊天都在线程里，天然互通）
   410→    const thread = loadMsgs(login.boundContactId).filter((m) => !m.kind || m.kind === 'text').slice(-14);
   411→    const history = thread.slice(0, -1).map((m) => ({ role: m.role === 'me' ? ('user' as const) : ('assistant' as const), content: m.content.slice(0, 800) }));
   412→    const contextText = thread.map((m) => `${m.role === 'me' ? '我' : 'TA'}: ${m.content}`).join('\n').slice(-1500);
   413→    const system = await buildSystemPrompt(login, contextText);
   414→    const reply = sanitizeWxOutText((await callLlm([...history, { role: 'user', content: text.slice(0, 2000) }])).trim());
   415→    if (!reply) throw new Error('生成了空回复');
   416→
   417→    await ilink('/ilink/bot/sendmessage', {
   418→      botToken: login.botToken,
   419→      host: api.host,
   420→      body: {
   421→        msg: {
   422→          from_user_id: '',
   423→          to_user_id: msg.from_user_id,
   424→          client_id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   425→          message_type: 2,
   426→          message_state: 2,
   427→          context_token: msg.context_token,
   428→          item_list: [{ type: 1, text_item: { text: reply.slice(0, 2000) } }],
   429→        },
   430→        base_info: BASE_INFO,
   431→      },
   432→    });
   433→    st.patchMsg(entry.id, { reply: reply.slice(0, 2000) });
   434→    // 角色说的 → 线程里记为「对方(peer)」，并标记为我们转发的（别再往微信发一遍）
   435→    const peerId = `wxr-${entry.id}-r`;
   436→    knownPeerIds.add(peerId);
   437→    appendThreadMsg(login.boundContactId, { id: peerId, role: 'peer', content: reply.slice(0, 2000), time: Date.now() });
   438→  } catch (e) {
   439→    st.patchMsg(entry.id, { replyErr: e instanceof Error ? e.message : String(e) });
   440→  } finally {
   441→    stopTyping();
   442→  }
   443→}

   445→// ---------------- 网页侧新回复 → 转发回真机微信 ----------------
   446→
   447→/** 已知「对方(peer)」消息 id 集合：我们自己落的不转发；boot 时把存量全部视为已知（不发历史） */
   448→const knownPeerIds = new Set<string>();
   449→
   450→async function forwardThreadWatcher(login: WxIlinkLogin): Promise<void> {
   451→  try {
   452→    const list = loadMsgs(login.boundContactId);
   453→    const fresh = list.filter((m) => m.role === 'peer' && (!m.kind || m.kind === 'text') && !knownPeerIds.has(m.id));
   454→    for (const m of fresh) {
   455→      knownPeerIds.add(m.id);
   456→      if (!lastCtx) continue; // 还没有任何微信来消息，没有 context_token 可用
   457→      await ilink('/ilink/bot/sendmessage', {
   458→        botToken: login.botToken,
   459→        host: hostOf(login.baseUrl),
   460→        body: {
   461→          msg: {
   462→            from_user_id: '',
   463→            to_user_id: lastCtx.fromUserId,
   464→            client_id: `${Date.now()}-${Math.random().toString(36).slice(2)}`,
   465→            message_type: 2,
   466→            message_state: 2,
   467→            context_token: lastCtx.contextToken,
   468→            item_list: [{ type: 1, text_item: { text: sanitizeWxOutText(m.content).slice(0, 2000) } }],
   469→          },
   470→          base_info: BASE_INFO,
   471→        },
   472→      }).catch(() => undefined);
   473→    }
   474→  } catch {
   475→    /* 转发尽力而为 */
   476→  }
   477→}

   479→// ---------------- 本机轮询主循环 ----------------
   480→
   481→async function runLoop(gen: number): Promise<void> {
   482→  const st = useWxConnect.getState();
   483→  st.setLoop('running', '');
   484→  while (gen === loopGen) {
   485→    const { login, autoReply } = useWxConnect.getState();
   486→    if (!login || !autoReply) break;
   487→    try {
   488→      const d = await ilink<{ ret?: number; error_code?: number; msgs?: ILinkMessage[]; get_updates_buf?: string }>(
   489→        '/ilink/bot/getupdates',
   490→        // host 跟随 confirmed 回包的 baseurl（换域后继续可用）；长轮询窗 50s
   491→        { botToken: login.botToken, host: hostOf(login.baseUrl), timeoutMs: 50_000, body: { get_updates_buf: updatesBuf, base_info: BASE_INFO } },
   492→      );
   493→      if (gen !== loopGen) break;
   494→      if (d.error_code === -14) {
   495→        useWxConnect.getState().setLoop('error', 'Token 已过期，请回到微信重新扫码接入');
   496→        return;
   497→      }
   498→      if (typeof d.get_updates_buf === 'string' && d.get_updates_buf) {
   499→        updatesBuf = d.get_updates_buf;
   500→        // v5：游标落盘（页面刷新后不重收积压——重复回复风暴的根因）
   501→        try {
   502→          kvSet(BUF_KEY, updatesBuf);
   503→        } catch {
   504→          /* 忽略 */
   505→        }
   506→      }
   507→      for (const m of d.msgs ?? []) {
   508→        if (gen !== loopGen) break;
   509→        await handleIncoming(useWxConnect.getState(), login, m).catch(() => undefined);
   510→      }
   511→      // 网页侧新回复转发（每轮一次）
   512→      await forwardThreadWatcher(login);
   513→      useWxConnect.setState({ lastPollAt: Date.now() });
   514→      if (useWxConnect.getState().loopStatus === 'error') {
   515→        useWxConnect.getState().setLoop('running', '');
   516→      }
   517→    } catch (e) {
   518→      if (gen !== loopGen) return;
   519→      useWxConnect.getState().setLoop('error', e instanceof Error ? e.message : String(e));
   520→      await sleep(5000);
   521→    }
   522→  }
   523→  if (gen === loopGen) useWxConnect.getState().setLoop('off', '');
   524→}

   528→interface WxConnectStore {
   529→  ready: boolean;
   530→  login: WxIlinkLogin | null;
   531→  autoReply: boolean;
   532→  loopStatus: 'off' | 'running' | 'error';
   533→  loopError: string;
   534→  lastPollAt: number | null;
   535→  msgs: WxIlinkMsg[];
   536→  qr: WxIlinkQr | null;
   537→  qrLoading: boolean;
   538→  testing: boolean;
   539→  lastTest: { ok: boolean; msg: string; at: number } | null;
   540→  /** 云端聊天记录拉取中（拉取云端代答的真机微信对话合进手机聊天界面） */
   541→  pulling: boolean;
   542→
   543→  boot: () => Promise<void>;
   544→  /** 选好绑定联系人后取二维码并开始官方协议状态机（长轮询 + PIN + 重定向 + 换码） */
   545→  startQr: (contactId: string, contactName: string) => Promise<void>;
   546→  /** 提交微信 PIN（need_verifycode 时 UI 输入 → 下一轮轮询带 verify_code） */
   547→  submitPin: (pin: string) => void;
   548→  cancelQr: () => void;
   549→  /** 换绑角色（不重新扫码，只改绑定的联系人） */
   550→  rebindContact: (contactId: string, contactName: string) => Promise<void>;
   551→  disconnect: () => Promise<void>;
   552→  /** 测试连接：getupdates 一次（有效 token 会快速返回，无效/过期会报错） */
   553→  test: () => Promise<boolean>;
   554→  setAutoReply: (v: boolean) => void;
   555→  pushMsg: (m: WxIlinkMsg) => void;
   556→  patchMsg: (id: string, patch: Partial<WxIlinkMsg>) => void;
   557→  clearMsgs: () => void;
   558→  /** 拉取云端微信聊天记录（测试云端/云端轮询代答的对话同步进手机聊天界面） */
   559→  pullHistory: () => Promise<void>;
   560→  setLoop: (s: 'off' | 'running' | 'error', err: string) => void;
   561→}
   562→
   563→/** 扫码状态机代数：start/stop/cancel 递增，旧循环自动退出 */
   564→let qrGen = 0;
   565→/** PIN 待提交的 resolver（need_verifycode 时 UI 输入后唤醒轮询循环） */
   566→let pinResolve: ((pin: string) => void) | null = null;
   567→
   568→export const useWxConnect = create<WxConnectStore>((set, get) => ({
   569→  ready: false,
   570→  login: null,
   571→  autoReply: false,
   572→  loopStatus: 'off',
   573→  loopError: '',
   574→  lastPollAt: null,
   575→  msgs: [],
   576→  qr: null,
   577→  qrLoading: false,
   578→  testing: false,
   579→  lastTest: null,
   580→  pulling: false,
   581→
   582→  boot: async () => {
   583→    if (wxConnectReady) return;
   584→    wxConnectReady = true;
   585→    try {
   586→      await ensureKvReady();
   587→      loadTokenHistory();
   588→      memLogin = await readEncJson<WxIlinkLogin>(LOGIN_KEY);
   589→      if (memLogin?.botToken) rememberToken(memLogin.botToken);
   590→      const msgs = await readEncJson<WxIlinkMsg[]>(MSGS_KEY).catch(() => null);
   591→      // v5：恢复游标（上次会话收过的不再重收——刷新页面后不会把积压重新回一遍）
   592→      try {
   593→        const savedBuf = kvGet<string>(BUF_KEY);
   594→        if (savedBuf) updatesBuf = savedBuf;
   595→      } catch {
   596→        /* 忽略 */
   597→      }
   598→      const auto = (() => {
   599→        try {
   600→          return kvGet<string>(AUTO_KEY) === '1';
   601→        } catch {
   602→          return false;
   603→        }
   604→      })();
   605→      const list = Array.isArray(msgs) ? msgs.slice(-MSGS_CAP) : [];
   606→      set({ login: memLogin, msgs: list, autoReply: auto && !!memLogin, ready: true });
   607→      if (memLogin) {
   608→        // 存量对方消息全部视为已知（不把历史回复重发到微信）
   609→        try {
   610→          for (const m of loadMsgs(memLogin.boundContactId)) {
   611→            if (m.role === 'peer') knownPeerIds.add(m.id);
   612→          }
   613→        } catch {
   614→          /* 忽略 */
   615→        }
   616→        if (auto) {
   617→          loopGen++;
   618→          void runLoop(loopGen);
   619→        }
   620→      }
   621→    } catch {
   622→      set({ ready: true });
   623→    }
   624→  },
   625→
   626→  /**
   627→   * 官方协议状态机（Tencent/openclaw-weixin docs/protocol_zh_CN.md）：
   628→   * ① POST get_bot_qrcode?bot_type=3，请求体 {local_token_list}（本机最近 10 个 bot token——
   629→   *   后端靠它认出环境亲缘，重绑不被当陌生环境风控）
   630→   * ② GET get_qrcode_status?qrcode=… 串行长轮询（服务端挂 ~35s；超时/网络错 = wait 续轮）。
   631→   *    状态：wait / scaned / confirmed（保存 bot_token+ilink_bot_id+baseurl+ilink_user_id）/
   632→   *    expired / 含 need_verifycode（手机显示数字 → 用户输入 → 下一轮带 verify_code，
   633→   *    再遇 need_verifycode = 输错重输）/ verify_code_blocked（错太多，换码重试 ≤3）/
   634→   *    scaned_but_redirect（后续轮询切 redirect_host）/ binded_redirect（已绑其它环境，终态引导）
   635→   * ③ 过期/锁死自动换码重试 3 次，超限提示稍后再试。
   636→   */
   637→  startQr: async (contactId, contactName) => {
   638→    if (get().qrLoading) return;
   639→    const myGen = ++qrGen;
   640→    set({ qrLoading: true, qr: null });
   641→    try {
   642→      await ensureKvReady();
   643→      // 换码重试外环（expired / verify_code_blocked ≤ 3 次）
   644→      for (let refresh = 0; refresh <= 3 && qrGen === myGen; refresh++) {
   645→        const d = await ilink<{ qrcode?: string; qrcode_img_content?: string; ret?: number }>(
   646→          '/ilink/bot/get_bot_qrcode?bot_type=3',
   647→          // local_token_list：本机 token 历史（协议：最多 10 个，后端认环境亲缘用）
   648→          { body: { local_token_list: tokenHistorySnapshot() } },
   649→        );
   650→        if (qrGen !== myGen) return;
   651→        if (!d.qrcode || !d.qrcode_img_content) throw new Error('iLink 未返回二维码');
   652→        // qrcode_img_content 是授权页 URL（不是图片）——本地生成二维码 data URL
   653→        const img = await QRCode.toDataURL(d.qrcode_img_content, { width: 320, margin: 2 });
   654→        set({ qr: { qrcode: d.qrcode, img, status: 'wait' }, qrLoading: false });
   655→
   656→        const qrcode = d.qrcode;
   657→        let statusHost: string | undefined; // scaned_but_redirect 后续切 redirect_host
   658→        let pin = ''; // 暂存 PIN：仅随「下一次」轮询上报（协议约定）
   659→        let wasPin = false; // 上一轮是否处于 PIN 输入（区分首次要求与输错重输）
   660→        interface QrStatusResp {
   661→          status?: string;
   662→          bot_token?: string;
   663→          ilink_bot_id?: string;
   664→          baseurl?: string;
   665→          ilink_user_id?: string;
   666→          redirect_host?: string;
   667→        }
   668→        // 串行长轮询内环
   669→        while (qrGen === myGen) {
   670→          const cur = get().qr;
   671→          if (!cur || cur.qrcode !== qrcode) return; // 被新码/取消取代
   672→          const qs = `qrcode=${encodeURIComponent(qrcode)}${pin ? `&verify_code=${encodeURIComponent(pin)}` : ''}`;
   673→          let s: QrStatusResp | null = null;
   674→          try {
   675→            s = await ilink<QrStatusResp>(`/ilink/bot/get_qrcode_status?${qs}`, {
   676→              method: 'GET',
   677→              body: {},
   678→              host: statusHost,
   679→              timeoutMs: 50_000,
   680→            });
   681→          } catch {
   682→            // 协议：单次失败/超时按 wait 续轮（短歇后重试）
   683→            await sleep(1000);
   684→            continue;
   685→          }
   686→          if (qrGen !== myGen) return;
   687→          const status = String(s?.status || 'wait');
   688→          // ① confirmed：保存凭证（含 baseurl / ilink_user_id）
   689→          if (status === 'confirmed' && s?.bot_token && s?.ilink_bot_id) {
   690→            const login: WxIlinkLogin = {
   691→              botToken: s.bot_token,
   692→              botId: s.ilink_bot_id,
   693→              ...(typeof s.baseurl === 'string' && s.baseurl ? { baseUrl: s.baseurl } : {}),
   694→              ...(typeof s.ilink_user_id === 'string' && s.ilink_user_id ? { ilinkUserId: s.ilink_user_id } : {}),
   695→              boundContactId: contactId,
   696→              boundContactName: contactName,
   697→              connectedAt: Date.now(),
   698→            };
   699→            memLogin = login;
   700→            await writeEncJson(LOGIN_KEY, login);
   701→            rememberToken(login.botToken);
   702→            personaCache = null;
   703→            lastCtx = null;
   704→            updatesBuf = '';
   705→            pinResolve = null;
   706→            set({ login, qr: { ...cur, status: 'confirmed', pinErr: undefined }, msgs: [], lastTest: null });
   707→            // 扫码接入成功后默认开启本机自动回复（用户可关；24h 在线建议部署云端）
   708→            get().setAutoReply(true);
   709→            // 已部署过云端则自动重推运行包（微信通道即刻生效，无需手动点同步）
   710→            cloudAutoSync('微信接入成功');
   711→            return;
   712→          }
   713→          // ② 需要输入 PIN（含 need_verifycode / confirmed_but_need_verifycode）
   714→          if (status.includes('need_verifycode')) {
   715→            set({
   716→              qr: {
   717→                ...cur,
   718→                status: 'need_verifycode',
   719→                pinErr: wasPin ? '数字不对，看看微信上显示的是多少，重新输一次' : undefined,
   720→              },
   721→            });
   722→            const typed = await new Promise<string>((resolve) => {
   723→              pinResolve = resolve;
   724→            });
   725→            if (qrGen !== myGen) return;
   726→            pin = typed.trim().replace(/\D/g, '').slice(0, 8);
   727→            wasPin = true;
   728→            continue; // 下一轮带 verify_code（返回 need_verifycode 依旧 = 输错）
   729→          }
   730→          // ③ 已扫码（带过 verify_code 时表示数字已接受 → 丢弃）
   731→          if (status === 'scaned') {
   732→            pin = '';
   733→            wasPin = false;
   734→            if (cur.status !== 'scaned') set({ qr: { ...cur, status: 'scaned', pinErr: undefined } });
   735→            continue;
   736→          }
   737→          // ④ 过期 / 错码锁死：换码重试（≤3），超限退出引导
   738→          if (status === 'expired' || status === 'verify_code_blocked') {
   739→            pin = '';
   740→            pinResolve = null;
   741→            if (refresh >= 3) {
   742→              set({
   743→                qr: {
   744→                  ...cur,
   745→                  status: 'expired',
   746→                  hint: status === 'verify_code_blocked' ? '数字错太多次被锁：稍后再试，或点重新获取' : undefined,
   747→                },
   748→              });
   749→              return;
   750→            }
   751→            break; // 外环换新码
   752→          }
   753→          // ⑤ 扫码后换域：后续轮询切 redirect_host
   754→          if (status === 'scaned_but_redirect') {
   755→            const rh = String(s?.redirect_host || '').trim();
   756→            if (rh) statusHost = rh.replace(/^https?:\/\//i, '').split('/')[0].toLowerCase();
   757→            if (cur.status !== 'scaned') set({ qr: { ...cur, status: 'scaned', pinErr: undefined } });
   758→            continue;
   759→          }
   760→          // ⑥ 已绑到其它环境（终态）：给解绑引导，停止轮询
   761→          if (status === 'binded_redirect') {
   762→            pinResolve = null;
   763→            set({
   764→              qr: {
   765→                ...cur,
   766→                status: 'binded_redirect',
   767→                hint: '这个微信号已连接到其它环境（比如之前连过别处的机器人）：先在那边断开/解绑，或换一个微信号扫码',
   768→              },
   769→            });
   770→            return;
   771→          }
   772→          // ⑦ wait：继续长轮询
   773→          if (cur.status !== 'wait') set({ qr: { ...cur, status: 'wait', pinErr: undefined } });
   774→        }
   775→      }
   776→    } catch (e) {
   777→      set({ qrLoading: false });
   778→      throw e;
   779→    }
   780→  },
   781→
   782→  submitPin: (pin) => {
   783→    const r = pinResolve;
   784→    pinResolve = null;
   785→    if (r) r(pin);
   786→  },
   787→
   788→  cancelQr: () => {
   789→    qrGen++;
   790→    const r = pinResolve;
   791→    pinResolve = null;
   792→    if (r) r('');
   793→    set({ qr: null });
   794→  },
   795→
   796→  rebindContact: async (contactId, contactName) => {
   797→    const login = get().login;
   798→    if (!login) return;
   799→    const next: WxIlinkLogin = { ...login, boundContactId: contactId, boundContactName: contactName };
   800→    memLogin = next;
   801→    personaCache = null;
   802→    await writeEncJson(LOGIN_KEY, next);

   923→// ---------------- 云端聊天记录拉取（云端代答的真机微信对话合进手机聊天界面） ----------------
   924→
   925→/** 云端 wx-cloud/history.json 的条目（角色视角：user=真机微信上发消息的人、assistant=TA 的回复） */
   926→interface WxCloudHistItem {
   927→  role?: string;
   928→  content?: string;
   929→  ts?: number;
   930→}
   931→
   932→/**
   933→ * 拉取云端微信聊天记录：云端助手代答（测试云端/定时轮询代跑）收发的对话存在 Supabase
   934→ * Storage wx-cloud/history.json（最近 30 条）——拉回来按「真机微信上的人=me、TA=peer」合进
   935→ * 手机里与绑定角色的微信聊天界面（去重：确定性 id，拉过的不会重复入列）。
   936→ * 与「本机实时回复」互不干扰：本机路径的消息已直接落聊天界面；这里只补云端路径的记录
   937→ * （用户实测反馈：点「测试云端」后 TA 在真机微信里回复了，但网站聊天界面里看不到——
   938→ * 云端代答的对话只有云端有，拉一次即合进来看）。
   939→ */
   940→export async function pullWxCloudHistory(): Promise<{ ok: boolean; msg: string; added?: number }> {
   941→  const cloud = wxCloudSnapshot();
   942→  const supa = cloud.supabase;
   943→  const login = memLogin;
   944→  if (!login) return { ok: false, msg: '还没接入真机微信' };
   945→  if (!login.boundContactId) return { ok: false, msg: '还没绑定角色：请先选择一个 AI 角色' };
   946→  if (!supa?.ref || !supa.serviceKey) return { ok: false, msg: '云端未就绪：请先到 设置 → 云端助手 完成部署' };
   947→  try {
   948→    // 新式 sb_secret_ key 只能放 apikey 头（Bearer 会被当 JWT 报 400 Invalid Compact JWS）
   949→    const k = supa.serviceKey;
   950→    const isNewStyle = k.startsWith('sb_secret_') || k.startsWith('sb_publishable_');
   951→    const res = await fetch(`https://${supa.ref}.supabase.co/storage/v1/object/wx-cloud/history.json`, {
   952→      headers: isNewStyle ? { apikey: k } : { apikey: k, Authorization: `Bearer ${k}` },
   953→      signal: AbortSignal.timeout(15_000),
   954→    });
   955→    if (!res.ok) return { ok: false, msg: '云端还没有微信聊天记录（真机微信上和 TA 聊过之后才有）' };
   956→    const j = (await res.json()) as { msgs?: WxCloudHistItem[] };
   957→    const list = Array.isArray(j.msgs) ? j.msgs : [];
   958→    const cur = loadMsgs(login.boundContactId);
   959→    const seen = new Set(cur.map((m) => m.id));
   960→    const add: WxMsg[] = [];
   961→    for (const m of list) {
   962→      if (!m || typeof m.content !== 'string' || !m.content.trim()) continue;
   963→      const mine = m.role === 'user'; // 云端视角：user=真机微信上发消息的人（=机主）→ 站内 me
   964→      const ts = Number(m.ts) || Date.now();
   965→      // 确定性 id：ts+角色+内容散列——同一云端记录重复拉取不重复入列（head 截断不撞号）
   966→      let h = 0;
   967→      const c = m.content.slice(0, 64);
   968→      for (let i = 0; i < c.length; i++) h = (h * 31 + c.charCodeAt(i)) >>> 0;
   969→      const id = `wxcloud-${ts}-${mine ? 'u' : 'a'}-${h % 9973}`;
   970→      if (seen.has(id)) continue;
   971→      add.push({ id, role: mine ? 'me' : 'peer', content: m.content.slice(0, 2000), time: ts });
   972→    }
   973→    if (add.length) {
   974→      add.sort((a, b) => a.time - b.time);
   975→      saveMsgs(login.boundContactId, [...cur, ...add]);
   976→      // 聊天页开着时实时增量合并（与工具轮系统行同款事件）
   977→      window.dispatchEvent(new CustomEvent('chat-stream-sysline', { detail: { sessionKey: `wx:${login.boundContactId}` } }));
   978→    }
   979→    return { ok: true, msg: add.length ? `已同步 ${add.length} 条到微信聊天界面` : '云端暂无新记录', added: add.length };
   980→  } catch (e) {
   981→    return { ok: false, msg: e instanceof Error ? e.message : String(e) };
   982→  }
   983→}
