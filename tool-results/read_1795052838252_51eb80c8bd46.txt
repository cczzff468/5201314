   291→      });

   294→      if (j && j.access_token) {
   295→        QQ_TOKEN = j.access_token;
   296→        QQ_TOKEN_AT = Date.now();

   299→      // 业务错误（官方约定 HTTP 200 + code）：换端点无意义，直接报可读错（code 兼容数字/字符串）
   300→      var bizCode = Number(j && j.code) || 0;
   301→      if (bizCode) {
   302→        if (bizCode === 100016) throw new Error('QQ 凭证不正确（100016）：请核对 AppID/AppSecret');
   303→        if (bizCode === 100007) throw new Error('QQ AppID 无效或机器人状态异常（100007）');
   304→        if (bizCode === 10004) throw new Error('QQ 机器人不存在（10004）');
   305→        if (bizCode === 100001) throw new Error('QQ 平台频控（100001）：稍后再试');
   306→        throw new Error('QQ token 获取失败 code ' + bizCode);
   307→      }
   308→      lastErr = 'HTTP ' + r.status;
   309→    } catch (te) {
   310→      if (String(te && te.message).indexOf('QQ ') === 0) throw te; // 业务错不再降级
   311→      lastErr = String(te && te.message);
   312→    }
   313→  }
   314→  throw new Error('QQ token 获取失败 ' + (lastStatus || '') + ' ' + lastErr.slice(0, 80));
   315→}

   408→/** 云端 TTS（OpenAI 兼容 /audio/speech，只要 wav）→ silk → QQ 原生语音条。失败返回 null（降级文字） */
   409→async function synthesizeQqVoice(cfg, text) {
   410→  var tts = cfg.qq && cfg.qq.tts;
   411→  if (!tts || !tts.baseUrl || !tts.apiKey) return null;
   412→  var b = String(tts.baseUrl).replace(/\/+$/, '');
   413→  var url = /\/audio\/speech$/.test(b) ? b : b + '/audio/speech';
   414→  try {
   415→    var r = await fetch(url, {
   416→      method: 'POST',
   417→      headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + tts.apiKey },
   418→      body: JSON.stringify({ model: tts.model || 'tts-1', voice: tts.voice || 'alloy', input: text.slice(0, 200), response_format: 'wav', speed: 1 }),
   419→      signal: AbortSignal.timeout(30000),
   420→    });
   421→    if (!r.ok) return null;
   422→    var buf = new Uint8Array(await r.arrayBuffer());
   423→    if (buf.length < 44 || String.fromCharCode(buf[0], buf[1], buf[2], buf[3]) !== 'RIFF') return null;
   424→    var silk = await silkEncode(buf);
   425→    if (!silk) return null;
   426→    var bin = '';
   427→    for (var i = 0; i < silk.length; i += 32768) bin += String.fromCharCode.apply(null, silk.subarray(i, i + 32768));
   428→    var b64 = (typeof btoa === 'function') ? btoa(bin) : Buffer.from(silk).toString('base64');
   429→    var dur = Math.max(1, Math.round(text.length / 4));
   430→    return { b64: b64, dur: dur };
   431→  } catch (e) { return null; }
   432→}

   676→async function edSign(seedStr, dataBytes) {
   677→  var key = await crypto.subtle.importKey('raw', seedBytes(seedStr), { name: 'Ed25519' }, false, ['sign']);
   678→  var sig = await crypto.subtle.sign({ name: 'Ed25519' }, key, dataBytes);
   679→  return new Uint8Array(sig);
   680→}
   681→
   682→async function edVerify(seedStr, dataBytes, sigBytes) {
   683→  try {
   684→    var key = await crypto.subtle.importKey('raw', seedBytes(seedStr), { name: 'Ed25519' }, false, ['verify']);
   685→    return await crypto.subtle.verify({ name: 'Ed25519' }, key, sigBytes, dataBytes);
   686→  } catch (e) { return false; }
   687→}
   688→
   689→function hexToBytes(hex) {
   690→  var out = new Uint8Array(hex.length / 2);
   691→  for (var i = 0; i < out.length; i++) out[i] = parseInt(hex.substr(i * 2, 2), 16);
   692→  return out;
   693→}
   694→
   695→function bytesToHex(b) {
   696→  var s = '';
   697→  for (var i = 0; i < b.length; i++) s += (b[i] < 16 ? '0' : '') + b[i].toString(16);
   698→  return s;
   699→}

   701→/** QQ Webhook 处理（Edge Function 入口调用）：校验请求 / 事件验签+代答。返回 Response */
   702→async function handleQqWebhook(cfg, req, rawBodyText) {
   703→  var seed = qqSeed(cfg.qq.appSecret);
   704→  if (!seed) return Response.json({ error: 'no secret' }, { status: 400 });
   705→  var payload = null;
   706→  try { payload = JSON.parse(rawBodyText); } catch (e) { payload = null; }
   707→  if (!payload || typeof payload !== 'object') return Response.json({ error: 'bad json' }, { status: 400 });
   708→  var d = payload.d || payload.data;
   709→  // 1) 校验请求：d 里带 plain_token/event_ts → 返回 {plain_token, signature}
   710→  if (d && typeof d === 'object' && d.plain_token) {
   711→    var msg = String(d.event_ts || '') + String(d.plain_token);
   712→    var sig = await edSign(seed, new TextEncoder().encode(msg));
   713→    return Response.json({ plain_token: d.plain_token, signature: bytesToHex(sig) });
   714→  }
   715→  // 2) 事件推送：验签 X-Signature-Ed25519（timestamp + body）
   716→  var sigHex = req.headers.get('x-signature-ed25519') || '';
   717→  var ts = req.headers.get('x-signature-timestamp') || '';
   718→  if (sigHex && ts) {
   719→    var data = new TextEncoder().encode(ts + rawBodyText);
   720→    var ok = await edVerify(seed, data, hexToBytes(sigHex));
   721→    if (!ok) return Response.json({ error: 'bad signature' }, { status: 401 });
   722→  }
   723→  var t = payload.t || payload.event_type || '';
   724→  if (t === 'C2C_MESSAGE_CREATE' || t === 'GROUP_AT_MESSAGE_CREATE') {
   725→    try {
   726→      await handleQqEvent(cfg, t, d);
   727→      return Response.json({ ok: true });

   729→      // 回复失败也回 200，避免平台反复重推同一条消息（历史里已去重）
   730→      return Response.json({ ok: false, error: String(e && e.message).slice(0, 200) });
   731→    }
   732→  }
   733→  return Response.json({ ok: true, ignored: t || 'unknown' });
   734→}
