   412→function cloudBurstSegments(text) {
   413→  var t = String(text || '');
   414→  var pieces = [];
   415→  var re = /\*([^*\n]{1,120})\*/g;
   416→  var last = 0;
   417→  var m = null;
   418→  while ((m = re.exec(t)) !== null) {
   419→    if (m.index > last) pieces.push({ act: false, text: t.slice(last, m.index) });
   420→    pieces.push({ act: true, text: m[1] });
   421→    last = m.index + m[0].length;
   422→  }
   423→  pieces.push({ act: false, text: t.slice(last) });
   424→  var out = [];
   425→  for (var ci = 0; ci < pieces.length; ci++) {
   426→    var p = pieces[ci];
   427→    if (p.act) {
   428→      var act = cloudTrim(p.text);
   429→      if (act) out.push(act);
   430→      continue;
   431→    }
   432→    var bRe = /&{3,}|\n|[。！？!?～〜]+[”’"』」）)\]]*/g;
   433→    var last2 = 0;
   434→    var mm = null;
   435→    while ((mm = bRe.exec(p.text)) !== null) {
   436→      var end = mm.index + mm[0].length;
   437→      var isSentence = mm[0] !== '\n' && mm[0].charAt(0) !== '&';
   438→      cloudPushSeg(out, p.text.slice(last2, isSentence ? end : mm.index));
   439→      last2 = end;
   440→    }
   441→    cloudPushSeg(out, p.text.slice(last2));
   442→  }
   443→  // 连续重复丢弃（不重复）
   444→  var dedup = [];
   445→  for (var di = 0; di < out.length; di++) {
   446→    if (di === 0 || out[di] !== out[di - 1]) dedup.push(out[di]);
   447→  }
   448→  out = dedup;
   449→  // 全员碎句合并（话少不拆）
   450→  if (out.length > 1) {
   451→    var allTiny = true;
   452→    for (var ti = 0; ti < out.length; ti++) { if (!cloudIsTiny(out[ti])) { allTiny = false; break; } }
   453→    if (allTiny) {
   454→      var joined = out[0];
   455→      for (var ji = 1; ji < out.length; ji++) joined = cloudJoinTiny(joined, out[ji]);
   456→      out = [joined];
   457→    }
   458→  }
   459→  return out;
   460→}
